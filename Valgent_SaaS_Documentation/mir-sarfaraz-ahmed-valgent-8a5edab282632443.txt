# Valgent

Valgent is an AI-powered Build & Validation Agent.

## Setup
1. Copy `.env.example` to `.env` and configure `GEMINI_API_KEY`.
2. Run `npm install`
3. Run `npm run build`
4. Run `npm run start`

## Sandbox Security Model
Valgent V0.2 uses a hardened Docker execution layer to execute untrusted repository code.

### Security Controls:
1. **No Host Access**: Does not mount host Docker socket or host home directories.
2. **Read-Only FS**: Uses `--read-only` root filesystem.
3. **Tmpfs Mounts**: Uses isolated `tmpfs` mounts for `/tmp` and `/home/node`.
4. **Limits**: Enforces CPU, Memory, and PID limits (`--cpus`, `--memory`, `--pids-limit`).
5. **No Privileges**: Drops all capabilities (`--cap-drop=ALL`) and sets `--security-opt no-new-privileges`.
6. **Network Policy**: Configurable network access (defaults to restricted `none`).
7. **User Isolation**: Runs as non-root `node` user (`1000:1000`).
8. **Command Abstraction**: Only pre-approved package manager operations (`npm run test`, `npm run build`, etc.) are permitted. No arbitrary host commands are executed.
9. **Environment Isolation**: No host environment variables (especially `GEMINI_API_KEY`) are passed into the container.
10. **Cleanup**: Ephemeral `--rm` containers are strictly force-removed upon execution completion or timeout.

## Tests
Run `npm run test`

import { ProjectMap } from '../analyzers/mapper';
import { RepositoryDescriptor, PackageManagerDetection, EnvironmentTelemetry } from '../shared/types';
import * as path from 'path';
import * as fs from 'fs';
import { DockerSandbox } from '../sandbox/docker';
import { EvidenceEngine } from '../evidence/engine';

export interface ResolvedEnvironment {
  telemetry: EnvironmentTelemetry;
  image: string;
}

export class EnvironmentResolver {
  private static readonly DEFAULT_NODE_VERSION = '20';
  private static readonly DEFAULT_BUN_VERSION = '1';

  public static async resolve(
    descriptor: RepositoryDescriptor,
    map: ProjectMap,
    packageManager: PackageManagerDetection,
    evidenceEngine: EvidenceEngine
  ): Promise<ResolvedEnvironment> {
    const { runtime, runtimeVersion, versionSource } = this.detectRuntime(descriptor.local_path, packageManager);

    let image = 'node:20-slim'; // Safe fallback
    if (runtime === 'bun') {
      image = `oven/bun:${runtimeVersion}`;
    } else {
      image = `node:${runtimeVersion}-slim`;
    }

    // Verify Environment using a temporary DockerSandbox
    const sandbox = new DockerSandbox(descriptor.local_path, image);
    
    let pmVersion = 'unknown';
    let resolutionStatus: 'RESOLVED' | 'FAILED' = 'FAILED';

    try {
      if (packageManager.name === 'bun') {
        const res = await sandbox.runCommand('bun --version', { timeoutMs: 10000 });
        if (res.exit_code === 0) {
          pmVersion = res.stdout.trim();
          resolutionStatus = 'RESOLVED';
        }
      } else if (packageManager.name === 'pnpm') {
        // Need to enable corepack first
        await sandbox.runCommand('corepack enable pnpm');
        const res = await sandbox.runCommand('corepack pnpm --version', { timeoutMs: 10000 });
        if (res.exit_code === 0) {
          pmVersion = res.stdout.trim();
          resolutionStatus = 'RESOLVED';
        }
      } else if (packageManager.name === 'yarn') {
        await sandbox.runCommand('corepack enable yarn');
        const res = await sandbox.runCommand('corepack yarn --version', { timeoutMs: 10000 });
        if (res.exit_code === 0) {
          pmVersion = res.stdout.trim();
          resolutionStatus = 'RESOLVED';
        }
      } else { // npm or unknown
        const res = await sandbox.runCommand('npm --version', { timeoutMs: 10000 });
        if (res.exit_code === 0) {
          pmVersion = res.stdout.trim();
          resolutionStatus = 'RESOLVED';
        }
      }
    } catch (err) {
      // Failed to execute verification command
    }

    const telemetry: EnvironmentTelemetry = {
      runtime,
      runtime_version: runtimeVersion,
      package_manager: packageManager.name,
      package_manager_version: pmVersion,
      image,
      resolution_status: resolutionStatus,
      source: versionSource
    };

    evidenceEngine.addEvidence(
      'ENVIRONMENT',
      versionSource,
      JSON.stringify(telemetry),
      `Resolved environment using ${versionSource}`,
      'repository',
      undefined,
      undefined,
      telemetry
    );

    return { telemetry, image };
  }

  private static detectRuntime(localPath: string, packageManager: PackageManagerDetection): { runtime: string, runtimeVersion: string, versionSource: string } {
    const isBun = packageManager.name === 'bun';
    const defaultRuntime = isBun ? 'bun' : 'node';
    const defaultVersion = isBun ? this.DEFAULT_BUN_VERSION : this.DEFAULT_NODE_VERSION;

    try {
      // 1. package.json engines or packageManager
      const pkgPath = path.join(localPath, 'package.json');
      if (fs.existsSync(pkgPath)) {
        const pkg = JSON.parse(fs.readFileSync(pkgPath, 'utf8'));
        if (pkg.engines && pkg.engines.node && !isBun) {
          const match = pkg.engines.node.match(/(\d+)/);
          if (match) {
            return { runtime: 'node', runtimeVersion: match[1], versionSource: 'package.json#engines.node' };
          }
        }
        if (pkg.engines && pkg.engines.bun && isBun) {
          const match = pkg.engines.bun.match(/(\d+)/);
          if (match) {
             return { runtime: 'bun', runtimeVersion: match[1], versionSource: 'package.json#engines.bun' };
          }
        }
        if (pkg.packageManager) {
          // e.g. "yarn@3.1.1", "pnpm@6.22.2" -> we can't easily get the node version from this, 
          // but if it's "bun@1.0.0", we could.
        }
      }

      // 2. .nvmrc
      const nvmrcPath = path.join(localPath, '.nvmrc');
      if (fs.existsSync(nvmrcPath) && !isBun) {
        const content = fs.readFileSync(nvmrcPath, 'utf8').trim();
        const match = content.match(/v?(\d+)/);
        if (match) {
          return { runtime: 'node', runtimeVersion: match[1], versionSource: '.nvmrc' };
        }
      }

      // 3. .node-version
      const nodeVersionPath = path.join(localPath, '.node-version');
      if (fs.existsSync(nodeVersionPath) && !isBun) {
        const content = fs.readFileSync(nodeVersionPath, 'utf8').trim();
        const match = content.match(/v?(\d+)/);
        if (match) {
          return { runtime: 'node', runtimeVersion: match[1], versionSource: '.node-version' };
        }
      }

      // 4. volta
      if (fs.existsSync(pkgPath)) {
        const pkg = JSON.parse(fs.readFileSync(pkgPath, 'utf8'));
        if (pkg.volta && pkg.volta.node && !isBun) {
           const match = pkg.volta.node.match(/(\d+)/);
           if (match) {
              return { runtime: 'node', runtimeVersion: match[1], versionSource: 'package.json#volta.node' };
           }
        }
      }
    } catch (err) {
      // ignore parsing errors
    }

    return { runtime: defaultRuntime, runtimeVersion: defaultVersion, versionSource: 'DEFAULT' };
  }
}

import { ProjectMap } from '../analyzers/mapper';
import { 
  RepositoryDescriptor, 
  ExecutionResult, 
  ExecutionPhaseResult,
  ExecutionPhaseName,
  ExecutionPhaseStatus,
  PackageManagerDetection,
  PackageManager,
  ContainerTelemetry
} from '../shared/types';
import { EvidenceEngine } from '../evidence/engine';
import { DockerSandbox, RawExecutionResult } from '../sandbox/docker';
import { EnvironmentResolver, ResolvedEnvironment } from './environment';
import * as crypto from 'crypto';

export interface ExecutionPolicy {
  allowDependencyInstall: boolean;
  allowBuild: boolean;
  allowTests: boolean;
  timeoutMs: number;
  memoryLimit: number;
  cpuLimit: number;
}

export class ExecutionOrchestrator {
  private policy: ExecutionPolicy;

  constructor(policy?: Partial<ExecutionPolicy>) {
    this.policy = {
      allowDependencyInstall: true,
      allowBuild: true,
      allowTests: true,
      timeoutMs: 120000,
      memoryLimit: 1024,
      cpuLimit: 2,
      ...policy
    };
  }

  public async orchestrate(descriptor: RepositoryDescriptor, map: ProjectMap, engine: EvidenceEngine): Promise<{ supported: boolean, telemetry?: ExecutionResult }> {
    if (!this.isSupported(map)) {
      return { supported: false };
    }

    const executionId = `EXE-${crypto.randomUUID()}`;
    const startedAt = new Date().toISOString();

    const pmDetection = this.detectPackageManager(map);
    
    // Environment Resolution
    const resolvedEnv = await EnvironmentResolver.resolve(descriptor, map, pmDetection, engine);
    const sandbox = new DockerSandbox(descriptor.local_path, resolvedEnv.image);

    const phases: ExecutionPhaseResult[] = [];
    
    // Phase 1: DISCOVERY
    phases.push({
      phase: 'DISCOVERY',
      status: 'PASSED',
      duration_ms: 0,
      network_enabled: false
    });

    let installFailed = false;
    let buildFailed = false;
    let envFailed = resolvedEnv.telemetry.resolution_status === 'FAILED';

    // Phase 2: INSTALL
    if (this.policy.allowDependencyInstall) {
      if (pmDetection.name === 'unknown') {
         phases.push({ phase: 'INSTALL', status: 'NOT_SUPPORTED', duration_ms: 0, network_enabled: true, skip_reason: 'No supported package manager detected' });
         installFailed = true;
      } else if (envFailed) {
         phases.push({ phase: 'INSTALL', status: 'SKIPPED', duration_ms: 0, network_enabled: true, skip_reason: 'Environment resolution failed' });
         installFailed = true;
      } else {
         const installCmd = this.getInstallCommand(pmDetection.name);
         // For pnpm and yarn, corepack must be enabled before install if it's node based
         let actualCmd = installCmd;
         if (pmDetection.name === 'pnpm' || pmDetection.name === 'yarn') {
             actualCmd = `corepack enable && corepack ${installCmd}`;
         }
         const res = await this.runPhase(sandbox, 'INSTALL', actualCmd, engine, true, executionId);
         phases.push(res);
         if (res.status !== 'PASSED') {
           installFailed = true;
         }
      }
    } else {
      phases.push({ phase: 'INSTALL', status: 'SKIPPED', duration_ms: 0, network_enabled: true, skip_reason: 'Policy disabled' });
    }

    // Phase 3: BUILD
    if (this.policy.allowBuild) {
      if (installFailed) {
         phases.push({ phase: 'BUILD', status: 'SKIPPED', duration_ms: 0, network_enabled: false, skip_reason: 'Prerequisite INSTALL failed or skipped' });
         buildFailed = true;
      } else if (!map.scripts || !map.scripts['build']) {
         phases.push({ phase: 'BUILD', status: 'NOT_SUPPORTED', duration_ms: 0, network_enabled: false, skip_reason: 'No build script defined in package.json' });
      } else if (envFailed) {
         phases.push({ phase: 'BUILD', status: 'SKIPPED', duration_ms: 0, network_enabled: false, skip_reason: 'Environment resolution failed' });
         buildFailed = true;
      } else {
         let buildCmd = `${pmDetection.name} run build`;
         if (pmDetection.name === 'pnpm' || pmDetection.name === 'yarn') {
             buildCmd = `corepack ${pmDetection.name} run build`;
         }
         const res = await this.runPhase(sandbox, 'BUILD', buildCmd, engine, false, executionId); // network disabled for build
         phases.push(res);
         if (res.status !== 'PASSED') {
           buildFailed = true;
         }
      }
    } else {
      phases.push({ phase: 'BUILD', status: 'SKIPPED', duration_ms: 0, network_enabled: false, skip_reason: 'Policy disabled' });
    }

    // Phase 4: TEST
    if (this.policy.allowTests) {
      if (installFailed || buildFailed) {
         phases.push({ phase: 'TEST', status: 'SKIPPED', duration_ms: 0, network_enabled: false, skip_reason: 'Prerequisite INSTALL or BUILD failed' });
      } else if (!map.scripts || !map.scripts['test'] || map.scripts['test'].includes('no test specified')) {
         phases.push({ phase: 'TEST', status: 'NOT_SUPPORTED', duration_ms: 0, network_enabled: false, skip_reason: 'No test script defined in package.json' });
      } else if (envFailed) {
         phases.push({ phase: 'TEST', status: 'SKIPPED', duration_ms: 0, network_enabled: false, skip_reason: 'Environment resolution failed' });
      } else {
         let testCmd = `${pmDetection.name} run test`;
         if (pmDetection.name === 'pnpm' || pmDetection.name === 'yarn') {
             testCmd = `corepack ${pmDetection.name} run test`;
         }
         const res = await this.runPhase(sandbox, 'TEST', testCmd, engine, false, executionId);
         phases.push(res);
      }
    } else {
      phases.push({ phase: 'TEST', status: 'SKIPPED', duration_ms: 0, network_enabled: false, skip_reason: 'Policy disabled' });
    }

    const durationMs = phases.reduce((acc, p) => acc + (p.duration_ms || 0), 0);
    const usesNetwork = phases.some(p => p.network_enabled);
    
    const containerTelemetry: ContainerTelemetry = {
      image: resolvedEnv.image,
      network_mode: usesNetwork ? 'mixed' : 'none',
      cpu_limit: this.policy.cpuLimit.toString(),
      memory_limit: this.policy.memoryLimit.toString(),
      pid_limit: '64'
    };

    const telemetry: ExecutionResult = {
      execution_id: executionId,
      started_at: startedAt,
      duration_ms: durationMs,
      package_manager: pmDetection,
      environment: resolvedEnv.telemetry,
      phases,
      container: containerTelemetry
    };

    return { supported: true, telemetry };
  }

  private isSupported(map: ProjectMap): boolean {
    return map.languages.includes('JavaScript') || map.languages.includes('TypeScript');
  }

  private detectPackageManager(map: ProjectMap): PackageManagerDetection {
    let warnings: string[] = [];
    let detected: PackageManager = 'unknown';
    let confidence: 'HIGH' | 'LOW' = 'LOW';
    let detectedFrom = 'heuristic';

    const pms = map.packageManagers.filter(pm => pm !== 'npm' || map.directories.includes('node_modules') || true); // Wait, mapping has duplicates?
    
    // Lockfile based exact match
    const hasYarnLock = map.packageManagers.includes('yarn');
    const hasPnpmLock = map.packageManagers.includes('pnpm');
    const hasBunLock = map.packageManagers.includes('bun');
    const hasNpmLock = map.packageManagers.includes('npm');

    let countLocks = (hasYarnLock ? 1 : 0) + (hasPnpmLock ? 1 : 0) + (hasBunLock ? 1 : 0) + (hasNpmLock ? 1 : 0);

    if (countLocks > 1) {
      warnings.push('PACKAGE_MANAGER_CONFLICT');
      // deterministic fallback order: pnpm > yarn > bun > npm
      if (hasPnpmLock) { detected = 'pnpm'; detectedFrom = 'pnpm-lock.yaml (conflict fallback)'; }
      else if (hasYarnLock) { detected = 'yarn'; detectedFrom = 'yarn.lock (conflict fallback)'; }
      else if (hasBunLock) { detected = 'bun'; detectedFrom = 'bun.lock (conflict fallback)'; }
      else { detected = 'npm'; detectedFrom = 'package-lock.json (conflict fallback)'; }
    } else if (countLocks === 1) {
      confidence = 'HIGH';
      if (hasPnpmLock) { detected = 'pnpm'; detectedFrom = 'pnpm-lock.yaml'; }
      else if (hasYarnLock) { detected = 'yarn'; detectedFrom = 'yarn.lock'; }
      else if (hasBunLock) { detected = 'bun'; detectedFrom = 'bun.lock'; }
      else { detected = 'npm'; detectedFrom = 'package-lock.json'; }
    } else {
      // fallback if package.json exists
      if (map.dependencies || map.scripts) {
        detected = 'npm';
        detectedFrom = 'package.json fallback';
      }
    }

    return {
      name: detected,
      confidence,
      detected_from: detectedFrom,
      warnings
    };
  }

  private getInstallCommand(pm: PackageManager): string {
    switch (pm) {
      case 'npm': return 'npm install --ignore-scripts';
      case 'yarn': return 'yarn install --ignore-scripts';
      case 'pnpm': return 'pnpm install --ignore-scripts';
      case 'bun': return 'bun install --ignore-scripts';
      default: return 'npm install --ignore-scripts';
    }
  }

  private async runPhase(sandbox: DockerSandbox, phase: ExecutionPhaseName, command: string, engine: EvidenceEngine, networkEnabled: boolean, executionId: string): Promise<ExecutionPhaseResult> {
    const options = {
      networkEnabled,
      timeoutMs: this.policy.timeoutMs,
      memoryLimitMb: this.policy.memoryLimit,
      cpus: this.policy.cpuLimit
    };

    const res = await sandbox.runCommand(command, options);
    
    let status: ExecutionPhaseStatus = 'PASSED';
    if (res.timed_out) {
      status = 'TIMED_OUT';
    } else if (res.exit_code !== 0) {
      status = 'FAILED';
    }

    // Generate evidence
    engine.addEvidence(
      'COMMAND_OUTPUT',
      `execution:${executionId}:${phase}`,
      `STDOUT:\n${res.stdout}\nSTDERR:\n${res.stderr}`.substring(0, 10000),
      `Execution of ${phase} completed with status ${status}`,
      'sandbox',
      undefined,
      undefined,
      {
        phase,
        command,
        exit_code: res.exit_code,
        duration_ms: res.duration_ms,
        timed_out: res.timed_out,
        network_enabled: res.network_enabled
      }
    );

    return {
      phase,
      status,
      command,
      exit_code: res.exit_code,
      duration_ms: res.duration_ms,
      stdout: res.stdout,
      stderr: res.stderr,
      timed_out: res.timed_out,
      network_enabled: res.network_enabled
    };
  }
}

export interface ProjectMap {

  frameworks: string[];
  packageManagers: string[];
  entryPoints: string[];
  dependencies: Record<string, string>;
  directories: string[];
  scripts: Record<string, string>;
}

export class ProjectMapper {
  static async map(descriptor: RepositoryDescriptor): Promise<ProjectMap> {
    const rootPath = descriptor.local_path;
    const map: ProjectMap = {
      languages: [],
      frameworks: [],
      packageManagers: [],
      entryPoints: [],
      dependencies: {},
      directories: [],
      scripts: {}
    };

    // Very basic mapping for V0.1
    const entries = await fs.readdir(rootPath, { withFileTypes: true });
    
    let hasPackageJson = false;

    for (const entry of entries) {
      if (entry.isDirectory()) {
        if (!['.git', 'node_modules'].includes(entry.name)) {
          map.directories.push(entry.name);
        }
      } else {
        if (entry.name === 'package.json') {
          hasPackageJson = true;
        }
        if (entry.name === 'yarn.lock') {
          if (!map.packageManagers.includes('yarn')) map.packageManagers.push('yarn');
        }
        if (entry.name === 'pnpm-lock.yaml') {
          if (!map.packageManagers.includes('pnpm')) map.packageManagers.push('pnpm');
        }
        if (entry.name === 'bun.lock' || entry.name === 'bun.lockb') {
          if (!map.packageManagers.includes('bun')) map.packageManagers.push('bun');
        }
        if (entry.name === 'package-lock.json') {
          if (!map.packageManagers.includes('npm')) map.packageManagers.push('npm');
        }
        if (entry.name.endsWith('.ts')) {
          if (!map.languages.includes('TypeScript')) map.languages.push('TypeScript');
        }
        if (entry.name.endsWith('.js')) {
          if (!map.languages.includes('JavaScript')) map.languages.push('JavaScript');
        }
        if (['index.js', 'server.js', 'main.js', 'app.js', 'index.ts', 'server.ts', 'main.ts', 'app.ts'].includes(entry.name)) {
          map.entryPoints.push(entry.name);
        }
      }
    }

    if (hasPackageJson) {
      try {
        const pkgContent = await fs.readFile(path.join(rootPath, 'package.json'), 'utf-8');
        const pkg = JSON.parse(pkgContent);
        if (pkg.scripts) {
          map.scripts = { ...pkg.scripts };
        }
        if (pkg.dependencies) {
          map.dependencies = { ...pkg.dependencies };
          
          // Detect frameworks from dependencies
          if (pkg.dependencies['express']) map.frameworks.push('Express');
          if (pkg.dependencies['react']) map.frameworks.push('React');
          if (pkg.dependencies['next']) map.frameworks.push('Next.js');
          if (pkg.dependencies['vue']) map.frameworks.push('Vue');
        }
      } catch (e) {
        console.error('Error parsing package.json', e);
      }
    }

    return map;
  }
}

export class StaticAnalyzer {
  static async analyze(descriptor: RepositoryDescriptor, map: ProjectMap, engine: EvidenceEngine): Promise<any> {
    const rootPath = descriptor.local_path;

    if (!map.languages.includes('JavaScript') && !map.languages.includes('TypeScript')) {
      engine.addEvidence(
        'COMMAND_OUTPUT',
        'static_analyzer',
        'unsupported_for_static_analysis',

        'analyzer'
      );
      return null;
    }

    // 1. Gather files
    const allFiles = await this.getAllFiles(rootPath, rootPath);
    
    // 2. Parse using semantic parser
    const { SemanticParser } = require('./semantic/parser');
    const parser = new SemanticParser(rootPath, allFiles);
    const semanticModel = parser.parse();

    // 3. Populate Evidence Engine with deterministic semantic evidence
    semanticModel.symbols.forEach((sym: any) => {
      engine.addEvidence('AST_SYMBOL', sym.file, JSON.stringify(sym), `Symbol: ${sym.name}`, 'semantic_parser', sym.line_start, sym.line_end, {
         evidence_id: sym.symbol_id,
         kind: sym.kind,
         exported: sym.exported
      });
    });

    semanticModel.imports.forEach((imp: any) => {
      // Imports may not have lines in our simple model
      engine.addEvidence('AST_IMPORT', imp.importer, JSON.stringify(imp), `Import from ${imp.source}`, 'semantic_parser', undefined, undefined, {
         source: imp.source,
         symbols: imp.symbols
      });
    });

    semanticModel.routes.forEach((route: any) => {
      engine.addEvidence('API_ROUTE', route.file, JSON.stringify(route), `Route: ${route.method} ${route.path}`, 'semantic_parser', route.line, route.line, {
         evidence_id: route.route_id,
         method: route.method,
         path: route.path,
         handler: route.handler,
         middleware: route.middleware
      });
    });

    semanticModel.calls.forEach((call: any) => {
      engine.addEvidence('AST_CALL', call.caller_file, JSON.stringify(call), `Call to ${call.target_name}`, 'semantic_parser', call.line, call.line, {
         target_name: call.target_name,
         resolution: call.resolution
      });
    });

    semanticModel.databaseOperations.forEach((db: any) => {
      engine.addEvidence('DATABASE_OPERATION', db.file, JSON.stringify(db), `DB ${db.type} operation`, 'semantic_parser', db.line, db.line, {
         evidence_id: db.operation_id,
         type: db.type
      });
    });

    semanticModel.components.forEach((comp: any) => {
      engine.addEvidence('REACT_COMPONENT', comp.file, JSON.stringify(comp), `React Component: ${comp.component}`, 'semantic_parser', undefined, undefined, {
         component: comp.component,
         hooks: comp.hooks
      });
    });

    return semanticModel;
  }

  private static async getAllFiles(currentPath: string, rootPath: string): Promise<string[]> {
    const entries = await fs.readdir(currentPath, { withFileTypes: true });
    let files: string[] = [];

    for (const entry of entries) {
      if (entry.isDirectory()) {
        if (['node_modules', '.git', 'dist', 'build'].includes(entry.name)) {
          continue;
        }
        const subFiles = await this.getAllFiles(path.join(currentPath, entry.name), rootPath);
        files = files.concat(subFiles);
      } else {
        files.push(path.relative(rootPath, path.join(currentPath, entry.name)));
      }
    }
    return files;
  }
}

import * as ts from 'typescript';
import crypto from 'crypto';
import { 
  SemanticProject, AstSymbol, AstImport, AstExport, AstCall, 
  ApiRoute, DbOperation, ReactComponent 
} from './types';
import fs from 'fs';
import path from 'path';

export class SemanticParser {
  private projectDir: string;
  private program: ts.Program;
  private checker: ts.TypeChecker;

  private files: string[] = [];
  private symbols: AstSymbol[] = [];
  private imports: AstImport[] = [];
  private exports: AstExport[] = [];
  private calls: AstCall[] = [];
  private routes: ApiRoute[] = [];
  private databaseOperations: DbOperation[] = [];
  private components: ReactComponent[] = [];
  
  constructor(projectDir: string, files: string[]) {
    this.projectDir = projectDir;
    
    const tsFiles = files.filter(f => f.endsWith('.ts') || f.endsWith('.tsx') || f.endsWith('.js') || f.endsWith('.jsx'));
    
    // Convert to absolute paths
    const absoluteFiles = tsFiles.map(f => path.join(this.projectDir, f));

    this.program = ts.createProgram(absoluteFiles, {
      target: ts.ScriptTarget.ES2022,
      module: ts.ModuleKind.CommonJS,
      allowJs: true,
      jsx: ts.JsxEmit.React
    });
    
    this.checker = this.program.getTypeChecker();
  }

  public parse(): SemanticProject {
    for (const sourceFile of this.program.getSourceFiles()) {
      // Ignore node_modules
      if (!sourceFile.isDeclarationFile && !sourceFile.fileName.includes('node_modules')) {
        const relativePath = path.relative(this.projectDir, sourceFile.fileName).replace(/\\/g, '/');
        this.files.push(relativePath);
        this.visitNode(sourceFile, sourceFile, relativePath);
      }
    }

    return {
      files: this.files,
      symbols: this.symbols,
      imports: this.imports,
      exports: this.exports,
      calls: this.calls,
      routes: this.routes,
      databaseOperations: this.databaseOperations,
      components: this.components,
      dependencies: []
    };
  }

  private visitNode(node: ts.Node, sourceFile: ts.SourceFile, relativePath: string) {
    this.extractImports(node, sourceFile, relativePath);
    this.extractSymbols(node, sourceFile, relativePath);
    this.extractCallsAndRoutes(node, sourceFile, relativePath);
    this.extractReactComponents(node, sourceFile, relativePath);
    this.extractDbOperations(node, sourceFile, relativePath);

    ts.forEachChild(node, child => this.visitNode(child, sourceFile, relativePath));
  }

  private getLineInfo(node: ts.Node, sourceFile: ts.SourceFile) {
    const start = sourceFile.getLineAndCharacterOfPosition(node.getStart());
    const end = sourceFile.getLineAndCharacterOfPosition(node.getEnd());
    return { line_start: start.line + 1, line_end: end.line + 1 };
  }

  private generateId(type: string, file: string, line: number, name: string): string {
    const hash = crypto.createHash('sha256').update(`${type}:${file}:${line}:${name}`).digest('hex');
    return `${type}-${hash.substring(0, 12)}`;
  }

  private isExported(node: ts.Node): boolean {
    return (
      (ts.getCombinedModifierFlags(node as ts.Declaration) & ts.ModifierFlags.Export) !== 0 ||
      (node.parent && node.parent.kind === ts.SyntaxKind.SourceFile)
    );
  }

  private extractImports(node: ts.Node, sourceFile: ts.SourceFile, relativePath: string) {
    if (ts.isImportDeclaration(node)) {
      if (node.moduleSpecifier && ts.isStringLiteral(node.moduleSpecifier)) {
        const source = node.moduleSpecifier.text;
        const importedSymbols: any[] = [];

        if (node.importClause) {
          if (node.importClause.name) {
            importedSymbols.push({ localName: node.importClause.name.text, importedName: 'default' });
          }
          if (node.importClause.namedBindings) {
            if (ts.isNamedImports(node.importClause.namedBindings)) {
              node.importClause.namedBindings.elements.forEach(el => {
                if (el.propertyName) {
                   importedSymbols.push({ localName: el.name.text, importedName: el.propertyName.text });
                } else {
                   importedSymbols.push({ localName: el.name.text, importedName: el.name.text });
                }
              });
            } else if (ts.isNamespaceImport(node.importClause.namedBindings)) {
              importedSymbols.push({ localName: node.importClause.namedBindings.name.text, importedName: '*' });
            }
          }
        }

        this.imports.push({
          source,
          importer: relativePath,
          symbols: importedSymbols
        });
      }
    }

    if (ts.isVariableStatement(node)) {
      node.declarationList.declarations.forEach(dec => {
        if (dec.initializer && ts.isCallExpression(dec.initializer) && ts.isIdentifier(dec.initializer.expression) && dec.initializer.expression.text === 'require') {
          const args = dec.initializer.arguments;
          if (args.length > 0 && ts.isStringLiteral(args[0])) {
            const source = args[0].text;
            const importedSymbols: any[] = [];
            
            if (ts.isIdentifier(dec.name)) {
               importedSymbols.push({ localName: dec.name.text, importedName: '*' });
            } else if (ts.isObjectBindingPattern(dec.name)) {
               dec.name.elements.forEach(el => {
                  if (ts.isIdentifier(el.name)) {
                     const localName = el.name.text;
                     const importedName = el.propertyName && ts.isIdentifier(el.propertyName) ? el.propertyName.text : localName;
                     importedSymbols.push({ localName, importedName });
                  }
               });
            }
            
            if (importedSymbols.length > 0) {
               this.imports.push({
                 source,
                 importer: relativePath,
                 symbols: importedSymbols
               });
            }
          }
        }
      });
    }
  }

  private extractSymbols(node: ts.Node, sourceFile: ts.SourceFile, relativePath: string) {
    let name = '';
    let kind: AstSymbol['kind'] | null = null;

    if (ts.isFunctionDeclaration(node) && node.name) {
      name = node.name.text;
      kind = 'function';
    } else if (ts.isClassDeclaration(node) && node.name) {
      name = node.name.text;
      kind = 'class';
    } else if (ts.isVariableStatement(node)) {
      node.declarationList.declarations.forEach(dec => {
        if (ts.isIdentifier(dec.name)) {
          const { line_start, line_end } = this.getLineInfo(node, sourceFile);
          const isExp = this.isExported(node);
          
          let subKind: AstSymbol['kind'] = 'variable';
          if (dec.initializer && (ts.isArrowFunction(dec.initializer) || ts.isFunctionExpression(dec.initializer))) {
             subKind = 'function';
          }

          this.symbols.push({
            symbol_id: this.generateId('SYM', relativePath, line_start, dec.name.text),
            name: dec.name.text,
            kind: subKind,
            file: relativePath,
            line_start,
            line_end,
            exported: isExp
          });
        }
      });
      return; // Handled
    }

    if (kind && name) {
      const { line_start, line_end } = this.getLineInfo(node, sourceFile);
      this.symbols.push({
        symbol_id: this.generateId('SYM', relativePath, line_start, name),
        name,
        kind,
        file: relativePath,
        line_start,
        line_end,
        exported: this.isExported(node)
      });
    }
  }

  private extractCallsAndRoutes(node: ts.Node, sourceFile: ts.SourceFile, relativePath: string) {
    if (ts.isCallExpression(node)) {
      const { line_start } = this.getLineInfo(node, sourceFile);
      let targetName = '';
      
      if (ts.isIdentifier(node.expression)) {
        targetName = node.expression.text;
      } else if (ts.isPropertyAccessExpression(node.expression)) {
        if (ts.isIdentifier(node.expression.expression)) {
           targetName = `${node.expression.expression.text}.${node.expression.name.text}`;
        } else {
           targetName = node.expression.name.text;
        }
      }

      if (targetName) {
        this.calls.push({
          caller_file: relativePath,
          line: line_start,
          target_name: targetName,
          resolution: 'UNRESOLVED', // Simple parser doesn't do deep cross-file inference
          arguments: node.arguments.map(a => a.getText(sourceFile))
        });

        // Express Route Detection
        const routeMethods = ['get', 'post', 'put', 'patch', 'delete'];
        const parts = targetName.split('.');
        if (parts.length === 2 && routeMethods.includes(parts[1])) {
          const method = parts[1].toUpperCase();
          const args = node.arguments;
          if (args.length >= 2 && ts.isStringLiteral(args[0])) {
            const pathUrl = args[0].text;
            const middleware: string[] = [];
            let handlerName: string | undefined;

            for (let i = 1; i < args.length; i++) {
              const arg = args[i];
              if (ts.isIdentifier(arg)) {
                if (i === args.length - 1) handlerName = arg.text;
                else middleware.push(arg.text);
              }
            }

            this.routes.push({
              route_id: this.generateId('ROUTE', relativePath, line_start, `${method}:${pathUrl}`),
              method,
              path: pathUrl,
              file: relativePath,
              line: line_start,
              middleware,
              handler: handlerName
            });
          }
        }
      }
    }
  }

  private extractDbOperations(node: ts.Node, sourceFile: ts.SourceFile, relativePath: string) {
    if (ts.isCallExpression(node)) {
       let targetName = '';
       if (ts.isPropertyAccessExpression(node.expression)) {
         targetName = node.expression.name.text;
       } else if (ts.isIdentifier(node.expression)) {
         targetName = node.expression.text;
       }
       
       const dbMethods = ['insert', 'update', 'delete', 'select', 'find', 'save', 'query', 'execute', 'run', 'get', 'all'];
       if (targetName && dbMethods.includes(targetName.toLowerCase())) {
          // Verify if it looks like a DB op (e.g. `db.insert`, `pool.query`)
          let isDbOp = false;
          if (ts.isPropertyAccessExpression(node.expression) && ts.isIdentifier(node.expression.expression)) {
             const caller = node.expression.expression.text.toLowerCase();
             if (['db', 'database', 'pool', 'client', 'sql', 'firestore', 'collection', 'query'].includes(caller)) {
                isDbOp = true;
             }
          }
          
          // Or string literal contains SQL
          if (node.arguments.length > 0 && ts.isStringLiteral(node.arguments[0])) {
             const query = node.arguments[0].text.toLowerCase();
             if (query.includes('insert into') || query.includes('select ') || query.includes('update ') || query.includes('delete from')) {
                isDbOp = true;
             }
          }

          if (isDbOp) {
             const { line_start } = this.getLineInfo(node, sourceFile);
             let type: DbOperation['type'] = 'UNKNOWN';
             if (targetName.toLowerCase() === 'insert' || targetName.toLowerCase() === 'save' || targetName.toLowerCase() === 'add') type = 'INSERT';
             if (targetName.toLowerCase() === 'select' || targetName.toLowerCase() === 'find') type = 'SELECT';
             if (targetName.toLowerCase() === 'update') type = 'UPDATE';
             if (targetName.toLowerCase() === 'delete') type = 'DELETE';
             
             if (type === 'UNKNOWN') {
                if (node.arguments.length > 0 && ts.isStringLiteral(node.arguments[0])) {
                   const query = node.arguments[0].text.toLowerCase();
                   if (query.includes('insert into')) type = 'INSERT';
                   else if (query.includes('select ')) type = 'SELECT';
                   else if (query.includes('update ')) type = 'UPDATE';
                   else if (query.includes('delete from')) type = 'DELETE';
                }
             }

             this.databaseOperations.push({
               operation_id: this.generateId('DB', relativePath, line_start, type),
               type,
               file: relativePath,
               line: line_start,
               target: 'database'
             });
          }
       }
    }
  }

  private extractReactComponents(node: ts.Node, sourceFile: ts.SourceFile, relativePath: string) {
    if (ts.isFunctionDeclaration(node) && node.name) {
       this.checkFunctionForReact(node, node.name.text, sourceFile, relativePath);
    } else if (ts.isVariableDeclaration(node) && ts.isIdentifier(node.name)) {
       if (node.initializer && (ts.isArrowFunction(node.initializer) || ts.isFunctionExpression(node.initializer))) {
          this.checkFunctionForReact(node.initializer, node.name.text, sourceFile, relativePath);
       }
    }
  }

  private checkFunctionForReact(node: ts.FunctionDeclaration | ts.ArrowFunction | ts.FunctionExpression, name: string, sourceFile: ts.SourceFile, relativePath: string) {
    // If it returns JSX, it's a React component
    if (name.match(/^[A-Z]/)) { // React components typically start with uppercase
       let returnsJsx = false;
       const hooks: string[] = [];

       const visit = (n: ts.Node) => {
         if (ts.isJsxElement(n) || ts.isJsxSelfClosingElement(n) || ts.isJsxFragment(n)) {
            returnsJsx = true;
         }
         if (ts.isCallExpression(n) && ts.isIdentifier(n.expression)) {
            if (n.expression.text.startsWith('use')) {
               hooks.push(n.expression.text);
            }
         }
         ts.forEachChild(n, visit);
       };

       visit(node);

       if (returnsJsx || relativePath.endsWith('.jsx') || relativePath.endsWith('.tsx')) {
         if (!this.components.find(c => c.component === name)) {
           this.components.push({
             component: name,
             file: relativePath,
             hooks: [...new Set(hooks)]
           });
         }
       }
    }
  }
}

const router = Router();

const AnalyzeRequestSchema = z.object({
  repository_url: z.string().url(),
  commit_sha: z.string().optional(),
  project_id: z.string().optional(),
  claims: z.array(z.object({
    claim_id: z.string(),
    text: z.string()
  })).optional(),
  analysis_options: z.object({
    skip_llm: z.boolean().optional(),
    skip_execution: z.boolean().optional(),
    timeout_ms: z.number().positive().optional()
  }).optional(),
  execution_policy: z.object({
    allow_install: z.boolean().optional(),
    allow_build: z.boolean().optional(),
    allow_tests: z.boolean().optional(),
    timeout_ms: z.number().positive().optional(),
    memory_limit_mb: z.number().positive().optional()
  }).optional(),
  llm_policy: z.object({
    preferred_providers: z.array(z.string()).optional(),
    timeout_ms: z.number().positive().optional()
  }).optional()
});

/**
 * Structured pipeline log. Prefixes every message with analysis_id.
 * Never logs secrets.
 */
function plog(analysisId: string, message: string): void {
  console.log(`[${analysisId}] ${redactSecrets(message)}`);
}

function perr(analysisId: string, message: string, error?: any): void {
  const errMsg = error ? `: ${redactSecrets(String(error?.message || error))}` : '';
  console.error(`[${analysisId}] ${redactSecrets(message)}${errMsg}`);
}

router.post('/analyze', async (req: Request, res: Response) => {
  const analysisId = crypto.randomUUID();
  const tracker = new PipelineTracker(analysisId, req.body?.repository_url || 'unknown', req.body?.commit_sha);
  let descriptor: any = null;

  tracker.enterStage('INITIALIZED');
  plog(analysisId, 'Starting analysis for request');

  try {
    // Validate request
    const body = AnalyzeRequestSchema.parse(req.body);

    const execPolicy = {
      allowDependencyInstall: body.execution_policy?.allow_install ?? true,
      allowBuild: body.execution_policy?.allow_build ?? true,
      allowTests: body.execution_policy?.allow_tests ?? true,
      timeoutMs: body.execution_policy?.timeout_ms ?? 120000,
      memoryLimit: body.execution_policy?.memory_limit_mb ?? 1024,
      cpuLimit: 2
    };
    const orchestrator = new ExecutionOrchestrator(execPolicy);

    const llmTimeoutMs = body.llm_policy?.timeout_ms ?? 30000;
    const providers = [];
    if (body.llm_policy?.preferred_providers) {
      for (const pref of body.llm_policy.preferred_providers) {
        if (pref === 'gemini') providers.push(new GeminiProvider(llmTimeoutMs));
        if (pref === 'openrouter') providers.push(new OpenRouterProvider(llmTimeoutMs));
      }
    }
    if (providers.length === 0) {
      providers.push(new GeminiProvider(llmTimeoutMs));
      providers.push(new OpenRouterProvider(llmTimeoutMs));
    }
    const llmRouter = new LLMRouter(providers);

    // 1. Fetch Repository
    tracker.enterStage('INGESTING');
    plog(analysisId, 'Fetching repository...');
    descriptor = await GithubFetcher.fetch(body.repository_url, body.commit_sha);
    plog(analysisId, `Fetched at ${descriptor.local_path} (commit: ${descriptor.commit_sha})`);

    const engine = new EvidenceEngine();

    // 2. Map Project
    tracker.enterStage('PROJECT_MAPPED');
    plog(analysisId, 'Mapping project...');
    const map = await ProjectMapper.map(descriptor);

    // 3. Extract Claims (record if present)
    if (body.claims && body.claims.length > 0) {
      tracker.enterStage('CLAIMS_EXTRACTED');
      plog(analysisId, `Extracted ${body.claims.length} claims from request`);
    }

    // 4. Document Analysis & Static Analysis
    tracker.enterStage('STATIC_ANALYSIS');
    plog(analysisId, 'Analyzing documentation & static analysis...');
    await DocumentAnalyzer.analyze(descriptor, engine);
    const semanticModel = await StaticAnalyzer.analyze(descriptor, map, engine);

    // 5. Dynamic Analysis (Execution)
    let executionResult: any = undefined;
    if (body.analysis_options?.skip_execution) {
      plog(analysisId, 'Skipping dynamic analysis (execution) per request options');
      executionResult = {
        supported: true,
        telemetry: {
          execution_id: `EXE-SKIPPED-${crypto.randomUUID().substring(0,8)}`,
          started_at: new Date().toISOString(),
          duration_ms: 0,
          package_manager: { name: 'unknown', confidence: 'LOW', detected_from: 'skip_execution', warnings: [] },
          container: { image: 'none', network_mode: 'none', cpu_limit: '0', memory_limit: '0', pid_limit: '0' },
          phases: [
            { phase: 'INSTALL', status: 'NOT_ATTEMPTED', duration_ms: 0, network_enabled: false, skip_reason: 'skip_execution=true' },
            { phase: 'BUILD', status: 'NOT_ATTEMPTED', duration_ms: 0, network_enabled: false, skip_reason: 'skip_execution=true' },
            { phase: 'TEST', status: 'NOT_ATTEMPTED', duration_ms: 0, network_enabled: false, skip_reason: 'skip_execution=true' }
          ]
        }
      };
    } else {
      tracker.enterStage('DYNAMIC_ANALYSIS');
      plog(analysisId, 'Performing dynamic analysis (execution)...');
      executionResult = await orchestrator.orchestrate(descriptor, map, engine);
      if (executionResult?.telemetry?.execution_id) {
        tracker.addExecutionId(executionResult.telemetry.execution_id);
      }
    }

    // 6. Build Evidence Graph
    tracker.enterStage('EVIDENCE_BUILT');
    plog(analysisId, 'Building evidence graph...');
    const { EvidenceGraphBuilder } = require('../evidence/graph-builder');
    const graphBuilder = new EvidenceGraphBuilder();
    const execResults = executionResult?.telemetry ? [executionResult.telemetry] : [];
    const evidenceGraphObj = graphBuilder.build(body.claims || [], engine.getEvidence(), semanticModel, execResults);
    const evidenceGraph = evidenceGraphObj.serialize();

    // Record evidence IDs
    for (const ev of engine.getEvidence()) {
      tracker.addEvidenceId(ev.evidence_id);
    }

    // 7. LLM Reasoning
    let analysis: any = {
      architecture: { components: [], entry_points: [], api_routes: [], databases: [] },
      features: [],
      warnings: []
    };
    let llmStatus: any = { status: 'SUCCESS', provider: llmRouter.lastUsedProvider || 'unknown', stage: 'LLM_SYNTHESIS' };
    let llmFailed = false;
    let claimsValidation: any = { validations: [], roadmap: [], audit_warnings: [], audit_trail: [] };

    if (body.analysis_options?.skip_llm) {
      plog(analysisId, 'LLM reasoning skipped by policy');
      llmFailed = true;
      llmStatus = {
        status: 'UNAVAILABLE',
        provider: 'none',
        stage: 'LLM_SYNTHESIS',
        error_code: 'SERVICE_UNAVAILABLE',
        retryable: false
      };
    } else {
      tracker.enterStage('LLM_SYNTHESIS');
      plog(analysisId, 'Sending to LLM for reasoning...');

      try {
        analysis = await llmRouter.analyzeEvidence(engine.getEvidence(), evidenceGraph);
        llmStatus.provider = llmRouter.lastUsedProvider;
        tracker.setLlmInfo(llmRouter.lastUsedProvider);

        if (body.claims && body.claims.length > 0) {
          tracker.enterStage('AUTONOMOUS_VALIDATION');
          plog(analysisId, 'Autonomous validation...');
          const valOrch = new ValidationOrchestrator(engine, descriptor.local_path, body.analysis_options?.skip_execution);
          claimsValidation = await valOrch.orchestrate(body.claims, evidenceGraphObj, llmRouter);
        }
      } catch (llmError: any) {
        perr(analysisId, 'LLM synthesis failed', llmError);
        llmFailed = true;

        const llmClass = classifyLlmError(llmError);
        llmStatus = {
          status: 'UNAVAILABLE',
          provider: 'none',
          stage: 'LLM_SYNTHESIS',
          error_code: llmClass.error_code,
          retryable: llmClass.retryable
        };

        // If we failed LLM, still run validation with deterministic evidence
        if (body.claims && body.claims.length > 0) {
          tracker.enterStage('AUTONOMOUS_VALIDATION');
          plog(analysisId, 'Autonomous validation (LLM degraded)...');
          const valOrch = new ValidationOrchestrator(engine, descriptor.local_path, body.analysis_options?.skip_execution);
          claimsValidation = await valOrch.orchestrate(body.claims, evidenceGraphObj, llmRouter);
        }
      }
    }

    // Record validation action IDs
    if (claimsValidation.audit_trail) {
      for (const action of claimsValidation.audit_trail) {
        if (action.decision_id) tracker.addValidationActionId(action.decision_id);
      }
    }

    // 8. Generate Report
    tracker.enterStage('REPORT_AUDITED');
    plog(analysisId, 'Generating report...');

    const analysisStartTime = tracker.getMetadata().started_at;
    const warnings = [...(analysis.warnings || []), ...(claimsValidation.audit_warnings || [])];

    const report: ProjectReport = {
      schema_version: "1.0",
      analysis_run_id: analysisId,
      project: {
        id: body.project_id || `PROJ-${crypto.createHash('sha256').update(body.repository_url).digest('hex').substring(0, 8)}`,
        name: body.repository_url.split('/').pop() || 'unknown',
        repository_url: descriptor.repository_url,
        repository_commit: descriptor.commit_sha || 'unknown',
        package_manager: (map.packageManagers && map.packageManagers.length > 0) ? map.packageManagers[0] : 'unknown',
        architecture: {
          components: analysis.architecture?.components || [],
          entry_points: map.entryPoints.length > 0 ? map.entryPoints : (analysis.architecture?.entry_points || []),
          api_routes: analysis.architecture?.api_routes || [],
          databases: analysis.architecture?.databases || [],
          dependencies: Object.keys(map.dependencies)
        },
        features: analysis.features || []
      },
      analysis: {
        started_at: analysisStartTime,
        completed_at: new Date().toISOString(),
        duration_ms: Date.now() - new Date(analysisStartTime).getTime(),
        status: llmFailed ? "INSUFFICIENT_EVIDENCE" : "COMPLETED",
        stages: tracker.getCompletedStageNames() as AnalysisStage[]
      },
      summary: body.claims && body.claims.length > 0 ? ValidationEngine.calculateSummary(claimsValidation.validations) : {
        claims_total: 0,
        verified: 0,
        partially_verified: 0,
        not_verified: 0,
        insufficient_evidence: 0
      },
      requirements: claimsValidation.validations || [],
      claims: claimsValidation.validations || [],
      evidence: engine.getEvidence(),
      evidence_graph: evidenceGraph,
      validation_actions: claimsValidation.audit_trail || [],
      roadmap: claimsValidation.roadmap || [],
      diagnostics: {
        warnings: warnings,
        errors: [],
        execution_failures: executionResult?.telemetry?.phases.filter((p: any) => p.status === 'FAILED').map((p: any) => p.stderr || p.stdout || '') || [],
        provider_failures: llmFailed ? [llmStatus.error_code] : [],
        unsupported_features: []
      },
      analysis_metadata: tracker.complete(),
      llm: llmStatus,
      execution: {
        supported: true,
        telemetry: executionResult?.telemetry
      }
    };

    // 9. Cleanup
    tracker.enterStage('COMPLETED');
    if (descriptor) {
      plog(analysisId, 'Cleaning up workspace...');
      await GithubFetcher.cleanup(descriptor);
    }

    plog(analysisId, 'Analysis completed');
    res.status(200).json({
      analysis_id: analysisId,
      status: "COMPLETED",
      report
    });

  } catch (error: any) {
    const stage = tracker.currentStage || 'INITIALIZED';
    tracker.failStage(error?.message || 'Unknown error');
    perr(analysisId, `Error at stage ${stage}`, error);

    if (descriptor) {
       await GithubFetcher.cleanup(descriptor).catch(() => {});
    }

    if (error instanceof z.ZodError) {
      const analysisError: AnalysisError = {
        code: 'INVALID_REQUEST',
        message: 'Request validation failed',
        stage: String(stage),
        retryable: false,
        details: (error as any).errors.map((e: any) => `${e.path.join('.')}: ${e.message}`)
      };
      return res.status(400).json({ error: analysisError });
    }

    const analysisError = classifyError(error, String(stage));
    res.status(500).json({
      error: analysisError,
      analysis_metadata: tracker.complete()
    });
  }
});

export default router;

import { Evidence, EvidenceType } from '../shared/types';
import * as crypto from 'crypto';

export class EvidenceEngine {
  private evidenceStore: Evidence[] = [];

  public addEvidence(
    type: EvidenceType,
    location: string,
    content: string,
    description: string,
    source: string,
    lineStart?: number,
    lineEnd?: number,
    metadata?: Record<string, any>
  ): Evidence {
    const contentHash = crypto.createHash('sha256').update(content).digest('hex');
    
    // Deterministic classification
    let classification: import('../shared/types').EvidenceClassification = 'UNCLASSIFIED';
    if (['AST_NODE', 'AST_ROUTE', 'API_ROUTE', 'SOURCE_CODE', 'DATABASE_OPERATION', 'DATABASE_SCHEMA', 'AST_SYMBOL', 'AST_IMPORT', 'AST_EXPORT', 'AST_CALL', 'REACT_COMPONENT'].includes(type)) classification = 'IMPLEMENTATION';
    else if (['BUILD_RESULT', 'COMMAND_OUTPUT'].includes(type)) classification = 'EXECUTION';
    else if (['TEST', 'TEST_RESULT'].includes(type)) classification = 'TEST'; // 'TEST' maps to TEST
    else if (['RUNTIME_TRACE'].includes(type)) classification = 'RUNTIME';
    else if (['DOCUMENTATION', 'DOCUMENT', 'README'].includes(type)) classification = 'DOCUMENTATION'; // DOCUMENT
    else if (['CONFIGURATION'].includes(type)) classification = 'CONFIGURATION';
    else if (['DEPENDENCY'].includes(type)) classification = 'DEPENDENCY';
    
    const evidence: Evidence = {
      evidence_id: metadata && metadata.evidence_id ? metadata.evidence_id : `EVD-${crypto.randomUUID()}`,
      type,
      location,
      content_hash: contentHash,
      description,
      source,
      line_start: lineStart,
      line_end: lineEnd,
      classification,
      metadata
    };

    this.evidenceStore.push(evidence);
    return evidence;
  }

  public getEvidence(): Evidence[] {
    return [...this.evidenceStore];
  }

  public clear(): void {
    this.evidenceStore = [];
  }
}

import { EvidenceGraph } from './graph';
import { EvidenceGraphData, GraphNode, EvidenceEdge, ClaimChain, CompletenessStatus, GraphNodeType, EvidenceEdgeType } from './graph-types';
import { Evidence, ValidationResult, Claim, ExecutionResult, ExecutionPhaseResult } from '../shared/types';
import { SemanticProject, AstSymbol, ApiRoute, DbOperation, AstCall } from '../analyzers/semantic/types';
import crypto from 'crypto';
import path from 'path';

export class EvidenceGraphBuilder {
  private graph: EvidenceGraph;

  constructor() {
    this.graph = new EvidenceGraph();
  }

  private resolveModulePath(baseFile: string, importPath: string, files: string[]): string | undefined {
    if (!importPath.startsWith('.')) return undefined; // Ignore node_modules
    
    // Normalize path separators to POSIX style as parser stores them that way
    const dir = path.posix.dirname(baseFile);
    let targetBase = path.posix.join(dir, importPath);
    if (targetBase.startsWith('./')) targetBase = targetBase.slice(2);
    
    const candidates = [
      targetBase,
      `${targetBase}.ts`,
      `${targetBase}.tsx`,
      `${targetBase}.js`,
      `${targetBase}.jsx`,
      `${targetBase}/index.ts`,
      `${targetBase}/index.js`
    ];

    for (const cand of candidates) {
      if (files.includes(cand)) {
        return cand;
      }
    }
    return undefined;
  }

  public build(
    claims: Claim[],
    evidenceList: Evidence[],
    semanticProject?: SemanticProject,
    executionResults?: ExecutionResult[]
  ): EvidenceGraph {
    
    // 1. Add Evidence Nodes
    for (const evd of evidenceList) {
      this.graph.addNode({
        id: evd.evidence_id,
        type: 'evidence',
        label: evd.description || `Evidence ${evd.type}`,
        source: evd.location,
        metadata: evd
      });
    }

    // 2. Add Claim & Requirement Nodes
    for (const claim of claims) {
      this.graph.addNode({
        id: claim.claim_id,
        type: 'claim',
        label: claim.text,
      });

      const reqId = `REQ-${crypto.createHash('sha256').update(claim.text).digest('hex').substring(0, 8)}`;
      this.graph.addNode({
        id: reqId,
        type: 'requirement',
        label: claim.text,
        metadata: { claim_id: claim.claim_id, statement: claim.text, status: 'UNVALIDATED' }
      });

      this.graph.addEdge({
        edge_id: `EDGE-${claim.claim_id}-${reqId}`,
        from: claim.claim_id,
        to: reqId,
        type: 'CLAIM_REQUIRES',
        confidence: 1.0,
        evidence_refs: []
      });
    }

    // 3. Add Semantic Project Nodes and Edges
    if (semanticProject) {
      // Build Symbol Indexes
      const fileExports = new Map<string, Map<string, AstSymbol>>();
      for (const sym of semanticProject.symbols) {
        if (!fileExports.has(sym.file)) fileExports.set(sym.file, new Map());
        fileExports.get(sym.file)!.set(sym.name, sym);
        if (sym.exported && !fileExports.get(sym.file)!.has('default')) {
           fileExports.get(sym.file)!.set('default', sym);
        }
      }

      const fileImports = new Map<string, Map<string, AstSymbol>>();
      for (const imp of semanticProject.imports) {
        const resolvedPath = this.resolveModulePath(imp.importer, imp.source, semanticProject.files);
        if (resolvedPath) {
          if (!fileImports.has(imp.importer)) fileImports.set(imp.importer, new Map());
          const targetExports = fileExports.get(resolvedPath);
          if (targetExports) {
            for (const symRef of imp.symbols) {
               let targetSym: AstSymbol | undefined;
               if (symRef.importedName === '*') {
                  targetSym = Array.from(targetExports.values())[0];
               } else {
                  targetSym = targetExports.get(symRef.importedName);
               }
               if (targetSym) {
                 fileImports.get(imp.importer)!.set(symRef.localName, targetSym);
               }
            }
          }
        }
      }

      // Symbols
      for (const sym of semanticProject.symbols) {
        this.graph.addNode({
          id: sym.symbol_id,
          type: 'symbol',
          label: sym.name,
          source: sym.file,
          metadata: sym
        });
      }

      // Routes
      for (const route of semanticProject.routes) {
        this.graph.addNode({
          id: route.route_id,
          type: 'route',
          label: `${route.method} ${route.path}`,
          source: route.file,
          metadata: route
        });

        if (route.handler) {
          let handlerSym = semanticProject.symbols.find(s => s.name === route.handler && s.file === route.file);
          if (!handlerSym) {
             handlerSym = fileImports.get(route.file)?.get(route.handler);
          }
          if (handlerSym) {
             this.graph.addEdge({
               edge_id: `EDGE-${route.route_id}-${handlerSym.symbol_id}`,
               from: route.route_id,
               to: handlerSym.symbol_id,
               type: 'ROUTE_HANDLES',
               confidence: 1.0,
               evidence_refs: []
             });
          }
        }
      }

      // Database Operations
      for (const db of semanticProject.databaseOperations) {
        this.graph.addNode({
          id: db.operation_id,
          type: 'databaseOperation',
          label: `${db.type} operation`,
          source: db.file,
          metadata: db
        });
      }
      
      for (const db of semanticProject.databaseOperations) {
        const parentSym = semanticProject.symbols.find(s => 
          s.file === db.file && s.line_start !== undefined && s.line_end !== undefined && 
          db.line >= s.line_start && db.line <= s.line_end
        );
        if (parentSym) {
           this.graph.addEdge({
             edge_id: `EDGE-${parentSym.symbol_id}-${db.operation_id}`,
             from: parentSym.symbol_id,
             to: db.operation_id,
             type: 'SYMBOL_PERSISTS',
             confidence: 1.0,
             evidence_refs: []
           });
        }
      }

      for (const call of semanticProject.calls) {
         const callerSym = semanticProject.symbols.find(s => 
           s.file === call.caller_file && s.line_start !== undefined && s.line_end !== undefined && 
           call.line >= s.line_start && call.line <= s.line_end
         );
         if (callerSym) {
            let targetSym = semanticProject.symbols.find(s => s.name === call.target_name && s.file === call.caller_file);
            if (!targetSym) targetSym = fileImports.get(call.caller_file)?.get(call.target_name);
            
            if (!targetSym && call.target_name.includes('.')) {
               const baseName = call.target_name.split('.')[0];
               targetSym = semanticProject.symbols.find(s => s.name === baseName && s.file === call.caller_file);
               if (!targetSym) targetSym = fileImports.get(call.caller_file)?.get(baseName);
            }

            if (targetSym) {
               this.graph.addEdge({
                 edge_id: `EDGE-${callerSym.symbol_id}-${targetSym.symbol_id}`,
                 from: callerSym.symbol_id,
                 to: targetSym.symbol_id,
                 type: 'SYMBOL_CALLS',
                 confidence: 1.0,
                 evidence_refs: []
               });
            }
         }
      }
    }

    // 4. Execution Evidence
    if (executionResults) {
      for (const exec of executionResults) {
        for (const phase of exec.phases) {
           if (phase.phase === 'TEST' && phase.status === 'PASSED') {
              const testNodeId = `TEST-${exec.execution_id}-${phase.phase}`;
              this.graph.addNode({
                id: testNodeId,
                type: 'test',
                label: `Test Execution`,
                metadata: phase
              });

              const exeNodeId = `EXE-${exec.execution_id}-${phase.phase}`;
              this.graph.addNode({
                id: exeNodeId,
                type: 'executionResult',
                label: `Execution Passed`,
                metadata: phase
              });

              this.graph.addEdge({
                edge_id: `EDGE-${testNodeId}-${exeNodeId}`,
                from: testNodeId,
                to: exeNodeId,
                type: 'EXECUTION_VALIDATES',
                confidence: 1.0,
                evidence_refs: []
              });
           }
        }
      }
    }

    // 5. Connect Evidence to Nodes
    for (const evd of evidenceList) {
       if (evd.type === 'API_ROUTE') {
          // Attempt to link to route node
          if (evd.metadata && evd.metadata.route_id) {
             this.graph.addEdge({
               edge_id: `EDGE-EVD-${evd.evidence_id}-${evd.metadata.route_id}`,
               from: evd.evidence_id,
               to: evd.metadata.route_id,
               type: 'EVIDENCE_LOCATES',
               confidence: 1.0,
               evidence_refs: [evd.evidence_id]
             });
          }
       }
       if (evd.type === 'DATABASE_OPERATION') {
          if (evd.metadata && evd.metadata.operation_id) {
             this.graph.addEdge({
               edge_id: `EDGE-EVD-${evd.evidence_id}-${evd.metadata.operation_id}`,
               from: evd.evidence_id,
               to: evd.metadata.operation_id,
               type: 'EVIDENCE_LOCATES',
               confidence: 1.0,
               evidence_refs: [evd.evidence_id]
             });
          }
       }
    }

    // 6. Claim -> Semantic Node resolution heuristic
    // For V0.7 we do a basic keyword match to link requirements to semantic nodes as a starting point.
    // Real implementation would use the AST analysis from V0.6 and LLM extraction to form exact edges.
    for (const claim of claims) {
       const reqNode = this.graph.getNodesByType('requirement').find(r => r.metadata?.claim_id === claim.claim_id);
       if (reqNode) {
          const keywords = claim.text.toLowerCase().split(' ').filter(k => k.length > 3);
          let foundConnection = false;
          let hasExecution = false;

          for (const route of this.graph.getNodesByType('route')) {
             if (keywords.some(k => route.label.toLowerCase().includes(k))) {
                this.graph.addEdge({
                  edge_id: `EDGE-${reqNode.id}-${route.id}`,
                  from: reqNode.id,
                  to: route.id,
                  type: 'REQUIREMENT_SUPPORTED_BY',
                  confidence: 0.5,
                  evidence_refs: []
                });
                foundConnection = true;
             }
          }

          // Check if there's test execution
          if (this.graph.getNodesByType('executionResult').length > 0) {
             hasExecution = true;
          }

          let completeness: CompletenessStatus = 'UNRESOLVED';
          if (foundConnection && hasExecution) {
             completeness = 'COMPLETE';
          } else if (foundConnection) {
             completeness = 'PARTIAL';
          }

          this.graph.addClaimChain({
             claim_id: claim.claim_id,
             requirement_id: reqNode.id,
             completeness,
             nodes: [],
             edges: []
          });
       }
    }

    return this.graph;
  }
}

import { EvidenceGraph } from './graph';
import { GraphNode, EvidenceEdge, EvidenceEdgeType, GraphNodeType } from './graph-types';

export interface GraphQueryOptions {
  maxDepth?: number;
  allowedEdgeTypes?: EvidenceEdgeType[];
  allowedNodeTypes?: GraphNodeType[];
  maxNodes?: number;
  maxEvidenceBytes?: number;
}

export class EvidenceGraphQuery {
  constructor(private graph: EvidenceGraph) {}

  public getClaimSubgraph(claimId: string, options?: GraphQueryOptions): { nodes: GraphNode[], edges: EvidenceEdge[], truncated: boolean } {
    const maxDepth = options?.maxDepth ?? 10;
    const maxNodes = options?.maxNodes ?? 100;
    const allowedEdges = options?.allowedEdgeTypes;
    const allowedNodes = options?.allowedNodeTypes;

    const nodes = new Map<string, GraphNode>();
    const edges = new Map<string, EvidenceEdge>();
    let truncated = false;

    const claimNode = this.graph.getNode(claimId);
    if (!claimNode) return { nodes: [], edges: [], truncated };
    if (allowedNodes && !allowedNodes.includes(claimNode.type)) return { nodes: [], edges: [], truncated };

    nodes.set(claimId, claimNode);

    // Bounded BFS
    const queue: { id: string, depth: number }[] = [{ id: claimId, depth: 0 }];

    while(queue.length > 0) {
      const { id, depth } = queue.shift()!;
      if (depth >= maxDepth) continue;
      if (nodes.size >= maxNodes) break;

      const outEdges = this.graph.getEdgesFrom(id);
      for (const e of outEdges) {
         if (allowedEdges && !allowedEdges.includes(e.type)) continue;
         
         const toNode = this.graph.getNode(e.to);
         if (!toNode) continue;
         if (allowedNodes && !allowedNodes.includes(toNode.type)) continue;

         edges.set(e.edge_id, e);
         if (!nodes.has(e.to)) {
            if (nodes.size >= maxNodes) break;
            nodes.set(e.to, toNode);
            queue.push({ id: e.to, depth: depth + 1 });
         }
      }
      
      // Look backward for evidence connecting to this node
      // E.g., EVIDENCE_LOCATES to -> current node
      const allEdges = this.graph.serialize().edges;
      for (const e of allEdges) {
          if (e.to === id && (e.type === 'EVIDENCE_LOCATES' || e.type === 'EXECUTION_VALIDATES')) {
             if (allowedEdges && !allowedEdges.includes(e.type)) continue;
             const fromNode = this.graph.getNode(e.from);
             if (!fromNode) continue;
             if (allowedNodes && !allowedNodes.includes(fromNode.type)) continue;
             
             edges.set(e.edge_id, e);
             if (!nodes.has(e.from)) {
                if (nodes.size >= maxNodes) break;
                nodes.set(e.from, fromNode);
                queue.push({ id: e.from, depth: depth + 1 });
             }
          }
      }
    }

    // Sort to maintain deterministic subgraph ordering
    let sortedNodes = Array.from(nodes.values()).sort((a, b) => a.id.localeCompare(b.id));
    let sortedEdges = Array.from(edges.values()).sort((a, b) => a.edge_id.localeCompare(b.edge_id));

    if (options?.maxEvidenceBytes) {
      let currentBytes = 0;
      const keepNodes: GraphNode[] = [];
      for (const node of sortedNodes) {
        const nodeBytes = Buffer.byteLength(JSON.stringify(node), 'utf8');
        if (currentBytes + nodeBytes > options.maxEvidenceBytes) {
          truncated = true;
          break;
        }
        currentBytes += nodeBytes;
        keepNodes.push(node);
      }
      sortedNodes = keepNodes;
      const validNodeIds = new Set(sortedNodes.map(n => n.id));
      sortedEdges = sortedEdges.filter(e => validNodeIds.has(e.from) && validNodeIds.has(e.to));
    }

    return { nodes: sortedNodes, edges: sortedEdges, truncated };
  }
}

import { EvidenceGraphData, GraphNode, EvidenceEdge, GraphNodeType, EvidenceEdgeType, ClaimChain, CompletenessStatus } from './graph-types';

export class EvidenceGraph {
  private nodes = new Map<string, GraphNode>();
  private edges = new Map<string, EvidenceEdge>();
  private claimChains = new Map<string, ClaimChain>();

  public addNode(node: GraphNode): void {
    this.nodes.set(node.id, node);
  }

  public getNode(id: string): GraphNode | undefined {
    return this.nodes.get(id);
  }

  public hasNode(id: string): boolean {
    return this.nodes.has(id);
  }

  public getNodesByType(type: GraphNodeType): GraphNode[] {
    return Array.from(this.nodes.values()).filter(n => n.type === type);
  }

  public addEdge(edge: EvidenceEdge): void {
    if (!edge.edge_id) {
      edge.edge_id = `EDGE-${edge.from}-${edge.to}-${edge.type}`;
    }
    this.edges.set(edge.edge_id, edge);
  }

  public getEdge(id: string): EvidenceEdge | undefined {
    return this.edges.get(id);
  }

  public getEdgesFrom(fromId: string): EvidenceEdge[] {
    return Array.from(this.edges.values()).filter(e => e.from === fromId);
  }

  public getEdgesTo(toId: string): EvidenceEdge[] {
    return Array.from(this.edges.values()).filter(e => e.to === toId);
  }

  public addClaimChain(chain: ClaimChain): void {
    this.claimChains.set(chain.claim_id, chain);
  }

  public getClaimChain(claimId: string): ClaimChain | undefined {
    return this.claimChains.get(claimId);
  }

  public serialize(): EvidenceGraphData {
    return {
      graph_version: '1.0',
      nodes: Array.from(this.nodes.values()),
      edges: Array.from(this.edges.values()),
      claim_chains: Array.from(this.claimChains.values())
    };
  }

  public clear(): void {
    this.nodes.clear();
    this.edges.clear();
    this.claimChains.clear();
  }
}

import { exec } from 'child_process';
import { promisify } from 'util';
import * as fs from 'fs/promises';
import * as path from 'path';
import * as os from 'os';
import { RepositoryDescriptor } from '../shared/types';

const execAsync = promisify(exec);

export class GithubFetcher {
  /**
   * Validates a GitHub repository URL.
   */
  static isValidUrl(url: string): boolean {
    const regex = /^https:\/\/github\.com\/[\w.-]+\/[\w.-]+$/;
    return regex.test(url);
  }

  /**
   * Fetches the repository to a temporary directory.
   */
  static async fetch(repository_url: string, commit_sha?: string): Promise<RepositoryDescriptor> {
    if (!this.isValidUrl(repository_url)) {
      throw new Error(`Invalid GitHub URL: ${repository_url}`);
    }

    const tempDir = await fs.mkdtemp(path.join(os.tmpdir(), 'valgent-'));
    
    try {
      // Clone the repository
      // depth 1 unless commit_sha is provided
      if (commit_sha) {
        await execAsync(`git clone ${repository_url} .`, { cwd: tempDir });
        await execAsync(`git checkout ${commit_sha}`, { cwd: tempDir });
      } else {
        await execAsync(`git clone --depth 1 ${repository_url} .`, { cwd: tempDir });
      }

      // Determine actual commit and branch
      const { stdout: actualCommit } = await execAsync('git rev-parse HEAD', { cwd: tempDir });
      
      let branch = 'main';
      try {
        const { stdout: actualBranch } = await execAsync('git rev-parse --abbrev-ref HEAD', { cwd: tempDir });
        branch = actualBranch.trim();
      } catch (e) {
        // Detached HEAD or similar
        branch = 'detached';
      }

      return {
        repository_url,
        commit_sha: actualCommit.trim(),
        local_path: tempDir,
        branch,
        fetched_at: new Date().toISOString()
      };
    } catch (error) {
      // Attempt cleanup on failure
      try {
        await fs.rm(tempDir, { recursive: true, force: true });
      } catch (cleanupError) {
        // Ignore cleanup errors
      }
      throw error;
    }
  }

  /**
   * Cleans up the temporary directory.
   */
  static async cleanup(descriptor: RepositoryDescriptor): Promise<void> {
    if (descriptor.local_path && descriptor.local_path.startsWith(os.tmpdir())) {
      try {
        await fs.rm(descriptor.local_path, { recursive: true, force: true });
      } catch (error) {
        console.error(`Failed to clean up workspace at ${descriptor.local_path}`, error);
      }
    }
  }
}

import { Evidence } from '../shared/types';
import { LLMProvider, LLMAnalysisResult, LLMValidationResult } from './provider';

export class LLMRouter implements LLMProvider {
  public readonly name = 'router';
  private providers: LLMProvider[];
  public lastUsedProvider: string = 'none';

  constructor(providers: LLMProvider[]) {
    if (!providers || providers.length === 0) {
      throw new Error('LLMRouter requires at least one provider');
    }
    this.providers = providers;
  }

  public async analyzeEvidence(evidenceStore: Evidence[], graph?: any): Promise<LLMAnalysisResult> {
    let lastError: any;

    for (const provider of this.providers) {
      try {
        const result = await provider.analyzeEvidence(evidenceStore, graph);
        this.lastUsedProvider = provider.name;
        return result;
      } catch (error) {
        console.warn(`[LLMRouter] ${provider.name} analyzeEvidence failed:`, error);
        lastError = error;
      }
    }

    console.error(`[LLMRouter] All providers failed for analyzeEvidence. Last error:`, lastError);
    throw lastError;
  }

  public async validateClaims(packages: any[]): Promise<LLMValidationResult> {
    let lastError: any;

    for (const provider of this.providers) {
      try {
        const result = await provider.validateClaims(packages);
        this.lastUsedProvider = provider.name;
        return result;
      } catch (error) {
        console.warn(`[LLMRouter] ${provider.name} validateClaims failed:`, error);
        lastError = error;
      }
    }

    console.error(`[LLMRouter] All providers failed for validateClaims. Last error:`, lastError);
    throw lastError;
  }
}

      '--security-opt', 'no-new-privileges',
      '--cap-drop=ALL',
      '--read-only',
      '--tmpfs', '/tmp:rw,noexec,nosuid,size=100m',
      '--tmpfs', '/home/node:rw,size=200m',
      '-v', `${this.localPath}:/workspace:rw`,
      '-w', '/workspace',
      '--user', '1000:1000',
      this.image,
      ...cmdArgs
    ];

    const startTime = Date.now();
    let stdout = '';
    let stderr = '';
    const MAX_OUTPUT = 100 * 1024; // 100 KB

    return new Promise((resolve) => {
      const sanitizedEnv = { ...process.env };
      delete sanitizedEnv.GEMINI_API_KEY;
      delete sanitizedEnv.OPENROUTER_API_KEY;

      const child = spawn('docker', dockerArgs, {
        stdio: ['ignore', 'pipe', 'pipe'],
        env: sanitizedEnv
      });

      let timeoutTimer: NodeJS.Timeout;
      let timedOut = false;

      timeoutTimer = setTimeout(async () => {
        timedOut = true;
        try {
          await execAsync(`docker rm -f ${containerName}`);
        } catch (e) {}
      }, timeoutMs);

      child.stdout.on('data', (chunk) => {
        if (stdout.length < MAX_OUTPUT) stdout += chunk.toString();
      });

      child.stderr.on('data', (chunk) => {
        if (stderr.length < MAX_OUTPUT) stderr += chunk.toString();
      });

      child.on('close', async (code) => {
        clearTimeout(timeoutTimer);
        const duration_ms = Date.now() - startTime;
        
        try {
           await execAsync(`docker rm -f ${containerName}`);
        } catch (e) {}

        resolve({
          exit_code: code === null ? 1 : code,
          duration_ms,
          stdout,
          stderr,
          timed_out: timedOut,
          image: this.image,
          network_enabled: !!options.networkEnabled
        });
      });

      child.on('error', async (err) => {
        clearTimeout(timeoutTimer);
        try {
           await execAsync(`docker rm -f ${containerName}`);
        } catch (e) {}
        
        resolve({
          exit_code: 1,
          duration_ms: Date.now() - startTime,
          stdout: '',
          stderr: err.message,
          timed_out: false,
          image: this.image,
          network_enabled: !!options.networkEnabled,
          error: err.message
        });
      });
    });
  }
}

export type AnalysisErrorCode =
  | 'INVALID_REQUEST'
  | 'REPOSITORY_FETCH_FAILED'
  | 'PROJECT_MAPPING_FAILED'
  | 'STATIC_ANALYSIS_FAILED'
  | 'EXECUTION_FAILED'
  | 'LLM_UNAVAILABLE'
  | 'VALIDATION_FAILED'
  | 'REPORT_AUDIT_FAILED'
  | 'INTERNAL_ERROR';

export interface AnalysisError {
  code: AnalysisErrorCode;
  message: string;
  stage: string;
  retryable: boolean;
  details?: string[];

/** Patterns that must never appear in logs or error messages. */
const SECRET_PATTERNS = [
  /(?:AIza|sk-|or-)[A-Za-z0-9_-]{20,}/g,                   // API keys
  /Bearer\s+[A-Za-z0-9._~+\/=-]+/gi,                       // Auth headers
  /(?:GEMINI|OPENROUTER|JWT|SECRET|TOKEN)_?(?:API)?_?KEY\s*=\s*\S+/gi, // env vars
  /eyJ[A-Za-z0-9_-]{10,}\.[A-Za-z0-9_-]{10,}\.[A-Za-z0-9_-]{10,}/g,   // JWTs
];

/**
 * Redacts secret values from a string.
 * Replaces matched patterns with [REDACTED].
 */
export function redactSecrets(input: string): string {
  let sanitized = input;
  for (const pattern of SECRET_PATTERNS) {
    sanitized = sanitized.replace(pattern, '[REDACTED]');
  }
  return sanitized;
}

/**
 * Classifies an error into an AnalysisErrorCode based on the stage and error content.
 */
export function classifyError(error: any, stage: string): AnalysisError {
  const rawMessage = error?.message || String(error);
  const message = redactSecrets(rawMessage);

  // Stage-specific classification
  switch (stage) {
    case 'INITIALIZED':
    case 'INGESTING':
      if (rawMessage.includes('Invalid GitHub URL') || rawMessage.includes('ZodError')) {
        return { code: 'INVALID_REQUEST', message, stage, retryable: false };
      }
      return { code: 'REPOSITORY_FETCH_FAILED', message, stage, retryable: true };

    case 'PROJECT_MAPPED':
      return { code: 'PROJECT_MAPPING_FAILED', message, stage, retryable: false };

    case 'STATIC_ANALYSIS':
      return { code: 'STATIC_ANALYSIS_FAILED', message, stage, retryable: false };

    case 'DYNAMIC_ANALYSIS':
    case 'EXECUTION_PLANNING':
      return { code: 'EXECUTION_FAILED', message, stage, retryable: false };

    case 'LLM_SYNTHESIS':
      return { code: 'LLM_UNAVAILABLE', message, stage, retryable: true };

    case 'AUTONOMOUS_VALIDATION':
    case 'CLAIMS_VALIDATED':
      return { code: 'VALIDATION_FAILED', message, stage, retryable: false };

    case 'REPORT_GENERATED':
    case 'REPORT_AUDITED':
      return { code: 'REPORT_AUDIT_FAILED', message, stage, retryable: false };

    default:
      return { code: 'INTERNAL_ERROR', message, stage, retryable: false };
  }
}

/**
 * Classifies an LLM error into a specific error_code for the LlmStatus.
 */
export function classifyLlmError(error: any): { error_code: string; retryable: boolean } {
  const msg = error?.message || '';
  
  if (msg.includes('429') || msg.includes('QUOTA') || msg.includes('Quota exceeded') || msg.includes('RESOURCE_EXHAUSTED')) {
    return { error_code: 'QUOTA_EXCEEDED', retryable: true };
  }
  if (msg.includes('503') || msg.includes('high demand') || msg.includes('UNAVAILABLE') || msg.includes('rate-limits')) {
    return { error_code: 'SERVICE_UNAVAILABLE', retryable: true };
  }
  if (msg.includes('401') || msg.includes('403') || msg.includes('API_KEY_INVALID')) {
    return { error_code: 'AUTHENTICATION_FAILED', retryable: false };
  }
  if (msg.includes('500')) {
    return { error_code: 'SERVICE_UNAVAILABLE', retryable: true };
  }
  if (msg.includes('timeout') || msg.includes('TIMEOUT') || msg.includes('ETIMEDOUT') || msg.includes('ECONNRESET')) {
    return { error_code: 'SERVICE_UNAVAILABLE', retryable: true };
  }
  // Zod errors or JSON parse errors
  if (error?.constructor?.name === 'ZodError' || msg.includes('Unexpected token') || msg.includes('JSON')) {
    return { error_code: 'INVALID_RESPONSE', retryable: false };
  }
  
  return { error_code: 'UNKNOWN', retryable: false };
}

/**
 * Generates a deterministic evidence ID based on content.
 * 
 * Same content at the same location with the same type produces the same ID.
 * This satisfies requirement 3: deterministic IDs for deterministic artifacts.
 */
export function deterministicEvidenceId(
  type: string,
  location: string,
  contentHash: string
): string {
  const input = `${type}:${location}:${contentHash}`;
  const hash = crypto.createHash('sha256').update(input).digest('hex').substring(0, 16);
  return `EVD-${hash}`;
}

  COMPLETED: [],
  ANALYSIS_FAILED: []

/**
 * Tracks pipeline state progression.
 * Enforces that stages cannot regress and records timing.
 */
export class PipelineTracker {
  private metadata: AnalysisMetadata;
  private currentStageIndex: number = -1;

  constructor(analysisId: string, repositoryUrl: string, commitRef?: string) {
    this.metadata = {
      analysis_id: analysisId,
      repository_url: repositoryUrl,
      commit_ref: commitRef,
      started_at: new Date().toISOString(),
      stages: [],
      execution_ids: [],
      evidence_ids: [],
      validation_action_ids: [],
      final_status: 'IN_PROGRESS'
    };
  }

  get analysisId(): string {
    return this.metadata.analysis_id;
  }

  get currentStage(): PipelineStage | undefined {
    if (this.metadata.stages.length === 0) return undefined;
    return this.metadata.stages[this.metadata.stages.length - 1].stage;
  }

  /**
   * Transitions to a new pipeline stage.
   * Completes the current stage if one is active.
   */
  enterStage(stage: PipelineStage): void {
    const current = this.currentStage;
    if (current) {
      if (current === 'COMPLETED' || current === 'ANALYSIS_FAILED') {
        throw {
          code: 'INTERNAL_ERROR',
          message: `Invalid pipeline transition from ${current} to ${stage}`,
          stage: current,
          retryable: false
        } as AnalysisError;
      }
      const allowed = VALID_TRANSITIONS[current] || [];
      if (!allowed.includes(stage) && stage !== 'ANALYSIS_FAILED') {
        throw {
          code: 'INTERNAL_ERROR',
          message: `Invalid pipeline transition from ${current} to ${stage}`,
          stage: current,
          retryable: false
        } as AnalysisError;
      }
    }

    // Complete the current stage if in progress
    this.completeCurrentStage('COMPLETED');

    this.metadata.stages.push({
      stage,
      started_at: new Date().toISOString(),
      status: 'IN_PROGRESS'
    });
  }

  /**
   * Marks the current stage as completed or failed.
   */
  completeCurrentStage(status: 'COMPLETED' | 'FAILED' | 'SKIPPED', error?: string): void {
    const current = this.metadata.stages[this.metadata.stages.length - 1];
    if (current && current.status === 'IN_PROGRESS') {
      current.completed_at = new Date().toISOString();
      current.duration_ms = new Date(current.completed_at).getTime() - new Date(current.started_at).getTime();
      current.status = status;
      if (error) {
        current.error = redactSecrets(error);
      }
    }
  }

  /**
   * Marks a stage as failed with an error message.
   */
  failStage(error: string): void {
    this.completeCurrentStage('FAILED', error);
    this.metadata.final_status = 'FAILED';
  }

  /**
   * Records an execution ID generated during analysis.
   */
  addExecutionId(id: string): void {
    this.metadata.execution_ids.push(id);
  }

  /**
   * Records an evidence ID generated during analysis.
   */
  addEvidenceId(id: string): void {
    this.metadata.evidence_ids.push(id);
  }

  /**
   * Records a validation action decision ID.
   */
  addValidationActionId(id: string): void {
    this.metadata.validation_action_ids.push(id);
  }

  /**
   * Records the LLM provider/model used for synthesis.
   */
  setLlmInfo(provider: string, model?: string): void {
    this.metadata.llm_provider = provider;
    if (model) this.metadata.llm_model = model;
  }

  /**
   * Completes the analysis and returns the full metadata.
   */
  complete(): AnalysisMetadata {
    this.completeCurrentStage('COMPLETED');
    this.metadata.completed_at = new Date().toISOString();
    this.metadata.duration_ms = new Date(this.metadata.completed_at).getTime() - new Date(this.metadata.started_at).getTime();
    if (this.metadata.final_status !== 'FAILED') {
      this.metadata.final_status = 'COMPLETED';
    }
    return { ...this.metadata };
  }

  /**
   * Returns the current metadata snapshot.
   */
  getMetadata(): AnalysisMetadata {
    return { ...this.metadata };
  }

  /**
   * Returns just the stage names for backwards compatibility with AnalysisStage[].
   */
  getCompletedStageNames(): string[] {
    return this.metadata.stages
      .filter(s => s.status === 'COMPLETED')
      .map(s => s.stage);
  }
}

    claims: Claim[],
    graph?: any
  ): { validations: ValidationResult[], roadmap: RemediationTask[], audit_warnings: string[] } {
    const validEvidenceIds = new Set(evidenceStore.map(e => e.evidence_id));
    if (graph) {
       (graph.nodes || []).forEach((n: any) => validEvidenceIds.add(n.id));
       (graph.edges || []).forEach((e: any) => validEvidenceIds.add(e.edge_id));
    }
    const validClaimIds = new Set(claims.map(c => c.claim_id));
    const audit_warnings: string[] = [];

    const auditedValidations = validations.map(v => {
      let status = v.status;
      
      // Verify claim exists
      if (!validClaimIds.has(v.claim_id)) {
        audit_warnings.push(`Validation references unknown claim: ${v.claim_id}`);
        // But we'll keep it as we might just ignore it later or return it
      }

      // Check evidence references
      const originalRefsCount = v.evidence_refs.length;
      const validRefs = v.evidence_refs.filter(ref => validEvidenceIds.has(ref));
      
      if (validRefs.length < originalRefsCount) {
        audit_warnings.push(`Claim ${v.claim_id} referenced non-existent evidence IDs. They have been stripped.`);
      }

      // Check graph references
      let originalGraphRefsCount = 0;
      let validGraphRefs: string[] = [];
      if (v.graph_refs) {
         originalGraphRefsCount = v.graph_refs.length;
         validGraphRefs = v.graph_refs.filter(ref => validEvidenceIds.has(ref));
         if (validGraphRefs.length < originalGraphRefsCount) {
           audit_warnings.push(`Claim ${v.claim_id} referenced non-existent graph IDs. They have been stripped.`);
         }
      }

      // If no valid evidence refs, and status isn't INSUFFICIENT_EVIDENCE
      if (validRefs.length === 0 && validGraphRefs.length === 0 && status !== 'INSUFFICIENT_EVIDENCE') {
        status = 'INSUFFICIENT_EVIDENCE';
        audit_warnings.push(`Claim ${v.claim_id} downgraded to INSUFFICIENT_EVIDENCE due to lack of valid evidence or graph references.`);
      }

      // 1. Invalid status mappings check
      const validStatuses = ['VERIFIED', 'PARTIALLY_VERIFIED', 'NOT_VERIFIED', 'INSUFFICIENT_EVIDENCE'];
      if (!validStatuses.includes(status)) {
        status = 'INSUFFICIENT_EVIDENCE';
        audit_warnings.push(`Claim ${v.claim_id} downgraded: Invalid status mapping '${v.status}'.`);
      }

      // Semantic Checks for Execution Evidence
      let hasFailedExecution = false;
      let hasSuccessfulTestExecution = false;
      let hasSuccessfulBuildExecution = false;
      let hasSuccessfulInstallExecution = false;
      let hasAnySourceCode = false;

      for (const ref of validRefs) {
        const ev = evidenceStore.find(e => e.evidence_id === ref);
        if (ev) {
          if (ev.type === 'SOURCE_CODE' || ev.type === 'API_ROUTE') {
            hasAnySourceCode = true;
          }
          if (ev.type === 'COMMAND_OUTPUT' && ev.metadata) {
            const md = ev.metadata;
            if (md.exit_code !== 0 || md.timed_out || md.status === 'SKIPPED' || md.status === 'NOT_SUPPORTED') {
              hasFailedExecution = true;
            }
            if (md.exit_code === 0 && md.phase === 'TEST') {
              hasSuccessfulTestExecution = true;
            }
            if (md.exit_code === 0 && md.phase === 'BUILD') {
              hasSuccessfulBuildExecution = true;
            }
            if (md.exit_code === 0 && md.phase === 'INSTALL') {
              hasSuccessfulInstallExecution = true;
            }
          }
        }
      }

      // Check 3: Fabricated execution claims
      // If the conclusion text claims successful execution (test/build/install) but no corresponding successful evidence is cited
      const conclusionLower = (v.conclusion || '').toLowerCase();
      const claimsExecution = conclusionLower.includes('execution completed') || 
                              conclusionLower.includes('successfully executed') || 
                              conclusionLower.includes('test execution completed successfully') ||
                              conclusionLower.includes('tests passed') ||
                              conclusionLower.includes('build succeeded');
      
      const hasAnySuccessfulExecution = hasSuccessfulBuildExecution || hasSuccessfulTestExecution || hasSuccessfulInstallExecution;
      
      if (status === 'VERIFIED' && claimsExecution && !hasAnySuccessfulExecution) {
        status = 'INSUFFICIENT_EVIDENCE';
        audit_warnings.push(`Claim ${v.claim_id} downgraded to INSUFFICIENT_EVIDENCE: fabricated execution claim detected (claims execution success but cites no successful execution evidence).`);
      }

      // Check 1: VERIFIED but execution failed
      if (status === 'VERIFIED' && hasFailedExecution) {
         status = 'PARTIALLY_VERIFIED';
         audit_warnings.push(`Claim ${v.claim_id} downgraded to PARTIALLY_VERIFIED: execution failed but claim was marked VERIFIED.`);
      }

      // Check 2: Citing only failed execution as proof implementation is absent
      if (status === 'NOT_VERIFIED' && hasFailedExecution && !hasAnySourceCode) {
         status = 'INSUFFICIENT_EVIDENCE';
         audit_warnings.push(`Claim ${v.claim_id} downgraded to INSUFFICIENT_EVIDENCE: cannot conclude implementation absence solely from failed execution.`);
      }

      // Check 4: Claims route exists but no API_ROUTE evidence cited
      let hasApiRoute = false;
      let hasDbOperation = false;

      for (const ref of validRefs) {
        const ev = evidenceStore.find(e => e.evidence_id === ref);
        if (ev) {
          if (ev.type === 'API_ROUTE') hasApiRoute = true;
          if (ev.type === 'DATABASE_OPERATION') hasDbOperation = true;
        }
      }

      const claimsRoute = conclusionLower.includes('route exists') || conclusionLower.includes('api endpoint') || conclusionLower.includes('endpoint is implemented');
      if (status === 'VERIFIED' && claimsRoute && !hasApiRoute) {
         status = 'INSUFFICIENT_EVIDENCE';
         audit_warnings.push(`Claim ${v.claim_id} downgraded to INSUFFICIENT_EVIDENCE: claims a route exists but cites no API_ROUTE evidence.`);
      }

      const claimsDbWrite = conclusionLower.includes('database write') || conclusionLower.includes('inserts into database') || conclusionLower.includes('saves to database') || conclusionLower.includes('database operation');
      if (status === 'VERIFIED' && claimsDbWrite && !hasDbOperation) {
         status = 'INSUFFICIENT_EVIDENCE';
         audit_warnings.push(`Claim ${v.claim_id} downgraded to INSUFFICIENT_EVIDENCE: claims database writes without supporting DATABASE_OPERATION evidence.`);
      }

      // Check 5: Documentation alone is NOT implementation
      let hasImplementationEvidence = false;
      let hasDocumentationEvidence = false;
      for (const ref of validRefs) {
         const ev = evidenceStore.find(e => e.evidence_id === ref);
         if (ev) {
            if (ev.classification === 'IMPLEMENTATION' || ev.type === 'SOURCE_CODE' || ev.type === 'API_ROUTE') hasImplementationEvidence = true;
            if (ev.classification === 'DOCUMENTATION' || ev.type === 'DOCUMENT') hasDocumentationEvidence = true;
         }
      }

      if (status === 'VERIFIED' && !hasImplementationEvidence && hasDocumentationEvidence) {
         status = 'INSUFFICIENT_EVIDENCE';
         audit_warnings.push(`Claim ${v.claim_id} downgraded to INSUFFICIENT_EVIDENCE: documentation alone cannot prove implementation.`);
      }

      // Check 6: VERIFIED but missing execution
      if (status === 'VERIFIED' && hasImplementationEvidence && !hasAnySuccessfulExecution) {
         // Some claims may just claim structural presence (e.g. "a route exists"), but for behavioral claims we need execution.
         // Let's rely on evaluator for deterministic evaluation. If the deterministic evaluation says execution ABSENT, we could downgrade.
         // Actually, Auditor doesn't have the deterministic evaluation here, but we can look at the evidence.
      }

      return {
        ...v,
        status,
        evidence_refs: validRefs,
        graph_refs: validGraphRefs
      };
    });

    const auditedRoadmap = roadmap.filter(task => {
      if (!validClaimIds.has(task.claim_id)) {
        audit_warnings.push(`Roadmap task ${task.task_id} dropped: references unknown claim ${task.claim_id}.`);
        return false;
      }
      return true;
    }).map(task => {
      const validRefs = task.evidence_refs.filter(ref => validEvidenceIds.has(ref));
      if (validRefs.length < task.evidence_refs.length) {
         audit_warnings.push(`Roadmap task ${task.task_id} referenced non-existent evidence IDs. They have been stripped.`);
      }
      return {
        ...task,
        evidence_refs: validRefs
      };
    });

    return { validations: auditedValidations, roadmap: auditedRoadmap, audit_warnings };
  }
}

import { Evidence, Claim, ValidationResult, RemediationTask, ProjectReport } from '../shared/types';
import { LLMProvider } from '../llm/provider';
import { ReportAuditor } from './auditor';
import { z } from 'zod';
import { EvidenceGraphQuery } from '../evidence/graph-query';

export class ValidationEngine {
  public static async validate(
    claims: Claim[],
    evidenceStore: Evidence[],
    provider: LLMProvider,
    graph?: any
  ): Promise<{ validations: ValidationResult[], roadmap: RemediationTask[], audit_warnings: string[] }> {
    if (!claims || claims.length === 0) {
      return { validations: [], roadmap: [], audit_warnings: [] };
    }

    try {
      const packages: any[] = [];
      
      let query: EvidenceGraphQuery | undefined;
      if (graph) {
         const { EvidenceGraph } = require('../evidence/graph');
         let graphInst = graph;
         if (!(graph instanceof EvidenceGraph)) {
            graphInst = new EvidenceGraph();
            if (graph.nodes) graph.nodes.forEach((n: any) => graphInst.addNode(n));
            if (graph.edges) graph.edges.forEach((e: any) => graphInst.addEdge(e));
         }
         query = new EvidenceGraphQuery(graphInst);
      }

      for (const claim of claims) {
         let subgraph = { nodes: [], edges: [], claim_chains: [], graph_version: '1.0' };
         let relEvidence: Evidence[] = [];

         if (query) {
            const sub = query.getClaimSubgraph(claim.claim_id, { maxDepth: 10, maxNodes: 100, maxEvidenceBytes: 50000 });
            subgraph.nodes = sub.nodes as any;
            subgraph.edges = sub.edges as any;
            (subgraph as any).truncated = sub.truncated;
            
            // extract evidence subset
            const evidenceIds = new Set<string>();
            sub.nodes.filter(n => n.type === 'evidence').forEach(n => evidenceIds.add(n.id));
            sub.edges.forEach(e => {
               if (e.evidence_refs) e.evidence_refs.forEach(r => evidenceIds.add(r));
            });
            
            relEvidence = evidenceStore.filter(e => evidenceIds.has(e.evidence_id));
         } else {
            relEvidence = evidenceStore;
         }

         const { ClaimEvidenceEvaluator } = require('./evaluator');
         const { GraphRoadmapGenerator } = require('./roadmap');

         const reqId = (graph?.getNodesByType ? graph.getNodesByType('requirement') : (graph?.nodes || [])).find((n: any) => n.metadata?.claim_id === claim.claim_id)?.id;
         
         const evaluation = ClaimEvidenceEvaluator.evaluate(claim, reqId, subgraph as any, relEvidence);
         const roadmap = GraphRoadmapGenerator.generate(claim, reqId, subgraph as any, evaluation, relEvidence);
         
         packages.push({
           claim,
           requirement_id: reqId,
           subgraph,
           evidence: relEvidence,
           deterministic_evaluation: evaluation,
           deterministic_roadmap: roadmap
         });
      }

      // 1. Ask provider to perform validation
      const providerResult = await provider.validateClaims(packages);
      
      const combinedRoadmap = [...providerResult.roadmap];
      packages.forEach(p => {
        if (p.deterministic_roadmap) {
           p.deterministic_roadmap.forEach((task: any) => combinedRoadmap.push(task));
        }
      });
      
      // 2. Audit the results to ensure NO hallucinated evidence IDs
      const audited = ReportAuditor.audit(providerResult.validations, combinedRoadmap, evidenceStore, claims, graph);
      
      return audited;
    } catch (error) {
      console.error('[ValidationEngine] Caught error:', error);
      // If provider fails, we degrade gracefully for all claims
      const validations: ValidationResult[] = claims.map(c => ({
        claim_id: c.claim_id,
        text: c.text,
        status: 'INSUFFICIENT_EVIDENCE',
        conclusion: 'Deterministic evidence was collected, but semantic synthesis was unavailable.',
        evidence_refs: [],
        graph_refs: [],
        limitations: [],
        missing_evidence: [],
        requirement_refs: [],
        checks: []
      }));

      return { validations, roadmap: [], audit_warnings: ['Provider validation failed, defaulting to INSUFFICIENT_EVIDENCE'] };
    }
  }

  public static calculateSummary(validations: ValidationResult[]) {
    const summary = {
      claims_total: validations.length,
      verified: 0,
      partially_verified: 0,
      not_verified: 0,
      insufficient_evidence: 0
    };

    validations.forEach(v => {
      if (v.status === 'VERIFIED') summary.verified++;
      else if (v.status === 'PARTIALLY_VERIFIED') summary.partially_verified++;
      else if (v.status === 'NOT_VERIFIED') summary.not_verified++;
      else if (v.status === 'INSUFFICIENT_EVIDENCE') summary.insufficient_evidence++;
    });

    return summary;
  }
}

import { Claim, Evidence, DeterministicEvaluation } from '../shared/types';
import { GraphNode, EvidenceEdge } from '../evidence/graph-types';

export class ClaimEvidenceEvaluator {
  public static evaluate(
    claim: Claim,
    requirementId: string | undefined,
    subgraph: { nodes: GraphNode[]; edges: EvidenceEdge[]; truncated: boolean },
    evidenceStore: Evidence[]
  ): DeterministicEvaluation {
    let implementation_status: 'PRESENT' | 'PARTIAL' | 'ABSENT' = 'ABSENT';
    let execution_status: 'PASSED' | 'FAILED' | 'NOT_ATTEMPTED' | 'ABSENT' = 'ABSENT';
    let test_status: 'PASSED' | 'FAILED' | 'NOT_ATTEMPTED' | 'ABSENT' = 'ABSENT';
    let contradiction_status: 'CONTRADICTED' | 'NONE' = 'NONE';
    let evidence_sufficiency: 'SUFFICIENT' | 'PARTIAL' | 'INSUFFICIENT' = 'INSUFFICIENT';
    const constraints: string[] = [];

    const evidenceMap = new Map(evidenceStore.map(e => [e.evidence_id, e]));

    let implementationNodes = 0;
    let docNodes = 0;

    for (const node of subgraph.nodes) {
      const ev = evidenceMap.get(node.id);
      if (ev) {
        if (ev.classification === 'IMPLEMENTATION') implementationNodes++;
        if (ev.classification === 'DOCUMENTATION') docNodes++;
      }
    }

    if (implementationNodes > 0) {
      implementation_status = implementationNodes >= 2 ? 'PRESENT' : 'PARTIAL';
    }

    let buildPassed = false;
    let buildFailed = false;
    let testPassed = false;
    let testFailed = false;

    for (const ev of evidenceStore) {
      if (ev.classification === 'EXECUTION' && ev.metadata?.phase === 'BUILD') {
        if (ev.metadata.exit_code === 0 || ev.metadata.status === 'PASSED') buildPassed = true;
        else buildFailed = true;
      }
      if (ev.classification === 'TEST' && ev.metadata?.phase === 'TEST') {
        if (ev.metadata.exit_code === 0 || ev.metadata.status === 'PASSED') testPassed = true;
        else testFailed = true;
      }
    }

    if (buildFailed) execution_status = 'FAILED';
    else if (buildPassed) execution_status = 'PASSED';
    else execution_status = 'NOT_ATTEMPTED';

    if (testFailed) test_status = 'FAILED';
    else if (testPassed) test_status = 'PASSED';
    else test_status = 'NOT_ATTEMPTED';

    if (implementation_status === 'PRESENT' && (execution_status === 'FAILED' || test_status === 'FAILED')) {
      contradiction_status = 'CONTRADICTED';
      constraints.push('Execution or test failed despite implementation presence.');
    }

    if (implementation_status === 'PRESENT' && execution_status === 'PASSED' && test_status === 'PASSED') {
      evidence_sufficiency = 'SUFFICIENT';
    } else if (implementation_status === 'PRESENT' || implementation_status === 'PARTIAL') {
      evidence_sufficiency = 'PARTIAL';
    }

    if (implementation_status === 'ABSENT' && docNodes > 0) {
      constraints.push('Documentation alone is NOT implementation evidence.');
    }

    if (implementation_status === 'PRESENT' && (execution_status as string === 'ABSENT' || execution_status === 'NOT_ATTEMPTED')) {
      constraints.push('Implementation exists but execution was not attempted or absent.');
    }

    return {
      implementation_status,
      execution_status,
      test_status,
      contradiction_status,
      evidence_sufficiency,
      deterministic_constraints: constraints
    };
  }
}

import { Claim, ValidationResult, ActionDecisionRecord, ValidationStatus, ValidationAction, RemediationTask } from '../shared/types';
import { LLMProvider } from '../llm/provider';
import { EvidenceGraph } from '../evidence/graph';
import { EvidenceGraphQuery } from '../evidence/graph-query';
import { ValidationPlanner } from './planner';
import { ValidationSafetyPolicy } from './safety';
import { ClaimEvidenceEvaluator } from './evaluator';
import { DockerSandbox } from '../sandbox/docker';
import { EvidenceEngine } from '../evidence/engine';
import { ValidationEngine } from './engine';
import * as crypto from 'crypto';

export class ValidationOrchestrator {
  private executedActions = new Set<string>();
  
  constructor(private engine: EvidenceEngine, private localPath: string, private skipExecution: boolean = false) {}
  
  public async orchestrate(
    claims: Claim[],
    graph: EvidenceGraph,
    provider: LLMProvider
  ): Promise<{ validations: ValidationResult[], audit_warnings: string[], audit_trail: ActionDecisionRecord[], roadmap: RemediationTask[] }> {
    const audit_trail: ActionDecisionRecord[] = [];
    const MAX_ACTIONS_PER_CLAIM = 3;
    const query = new EvidenceGraphQuery(graph);
    
    let cachedBuildStatus = 'NOT_ATTEMPTED';
    
    for (const claim of claims) {
      let actionsTaken = 0;
      let claimVerified = false;
      
      while (actionsTaken < MAX_ACTIONS_PER_CLAIM && !claimVerified) {
        const sub = query.getClaimSubgraph(claim.claim_id, { maxDepth: 10, maxNodes: 100 });
        
        const reqId = graph.getNodesByType('requirement').find(n => n.metadata?.claim_id === claim.claim_id)?.id;
        const relEvidence = this.engine.getEvidence().filter(e => sub.nodes.some(n => n.id === e.evidence_id) || sub.edges.some(edge => edge.evidence_refs?.includes(e.evidence_id)));
        const evaluation = ClaimEvidenceEvaluator.evaluate(claim, reqId, sub as any, relEvidence);
        
        if (evaluation.evidence_sufficiency === 'SUFFICIENT') {
           claimVerified = true;
           break;
        }
        
        const candidates = ValidationPlanner.plan(claim, reqId || '', sub as any, evaluation);
        
        if (candidates.length === 0) {
           audit_trail.push({
             decision_id: crypto.randomUUID(),
             claim_id: claim.claim_id,
             candidate_actions: [],
             selection_reason: 'No useful action exists',
             safety_decision: 'REJECTED',
             result: 'ACTION_NOT_SELECTED',
             new_evidence: []
           });
           break; 
        }
        
        let selectedAction: ValidationAction | undefined = undefined;
        for (const c of candidates) {
            const dedupeKey = `${c.type}:${c.target || ''}`;
            if (!this.executedActions.has(dedupeKey)) {
                selectedAction = c;
                this.executedActions.add(dedupeKey);
                break;
            }
        }
        
        if (!selectedAction) {
           audit_trail.push({
             decision_id: crypto.randomUUID(),
             claim_id: claim.claim_id,
             candidate_actions: candidates,
             selection_reason: 'All candidate actions have already been executed',
             safety_decision: 'REJECTED',
             result: 'ACTION_SKIPPED_CACHED',
             new_evidence: []
           });
           break;
        }

        if (selectedAction.type === 'BUILD' && cachedBuildStatus === 'PASSED') {
           audit_trail.push({
             decision_id: crypto.randomUUID(),
             claim_id: claim.claim_id,
             candidate_actions: candidates,
             selected_action: selectedAction,
             selection_reason: 'Build cached',
             safety_decision: 'ALLOWED',
             result: 'ACTION_SKIPPED_CACHED',
             new_evidence: []
           });
           continue;
        }
        
        const safety = ValidationSafetyPolicy.evaluate(selectedAction);
        if (safety.status === 'REJECTED') {
           audit_trail.push({
             decision_id: crypto.randomUUID(),
             claim_id: claim.claim_id,
             candidate_actions: candidates,
             selected_action: selectedAction,
             selection_reason: safety.reason || 'Safety rejected',
             safety_decision: 'REJECTED',
             result: 'ACTION_REJECTED',
             new_evidence: []
           });
           break;
        }
        
        let resultStatus: 'ACTION_FAILED' | 'ACTION_SUCCEEDED' = 'ACTION_FAILED';
        const newEvIds: string[] = [];
        
        if (safety.command) {
            if (this.skipExecution) {
               audit_trail.push({
                 decision_id: crypto.randomUUID(),
                 claim_id: claim.claim_id,
                 candidate_actions: candidates,
                 selected_action: selectedAction,
                 selection_reason: 'Execution skipped per request options',
                 safety_decision: 'REJECTED',
                 result: 'ACTION_NOT_ATTEMPTED',
                 new_evidence: []
               });
               break;
            }

            const sandbox = new DockerSandbox(this.localPath);
            const executionId = `EXE-${crypto.randomUUID()}`;
            const runRes = await sandbox.runCommand(safety.command, safety.sandboxOptions);
            
            resultStatus = runRes.exit_code === 0 ? 'ACTION_SUCCEEDED' : 'ACTION_FAILED';
            if (selectedAction.type === 'BUILD') cachedBuildStatus = resultStatus === 'ACTION_SUCCEEDED' ? 'PASSED' : 'FAILED';
            
            const evType = selectedAction.type === 'BUILD' ? 'BUILD_RESULT' : (selectedAction.type === 'STATIC_INSPECTION' ? 'SOURCE_CODE' : 'TEST_RESULT');
            const ev = this.engine.addEvidence(
               evType as any,
               'sandbox',
               runRes.stdout + '\n' + runRes.stderr,
               `${selectedAction.type} output`,
               'execution',
               undefined, undefined,
               { execution_id: executionId, exit_code: runRes.exit_code, phase: selectedAction.type }
            );
            
            newEvIds.push(ev.evidence_id);
            graph.addNode({ id: ev.evidence_id, type: 'evidence', label: ev.evidence_id, metadata: { classification: ev.classification } });
            graph.addEdge({ edge_id: `E-${crypto.randomUUID()}`, from: claim.claim_id, to: ev.evidence_id, type: 'CLAIM_REQUIRES', confidence: 1, evidence_refs: [] });
        }
        
        const subAfter = query.getClaimSubgraph(claim.claim_id, { maxDepth: 10, maxNodes: 100 });
        const relEvidenceAfter = this.engine.getEvidence().filter(e => subAfter.nodes.some(n => n.id === e.evidence_id) || subAfter.edges.some(edge => edge.evidence_refs?.includes(e.evidence_id)));
        const evaluationAfter = ClaimEvidenceEvaluator.evaluate(claim, reqId, subAfter as any, relEvidenceAfter);

        let reEvalStatus: ValidationStatus = 'NOT_VERIFIED';
        if (evaluationAfter.evidence_sufficiency === 'SUFFICIENT') reEvalStatus = 'VERIFIED';
        else if (evaluationAfter.implementation_status === 'PARTIAL' || evaluationAfter.implementation_status === 'PRESENT') reEvalStatus = 'PARTIALLY_VERIFIED';
        if (evaluationAfter.contradiction_status === 'CONTRADICTED') reEvalStatus = 'NOT_VERIFIED';
        if (evaluationAfter.implementation_status === 'ABSENT' && evaluationAfter.execution_status === 'NOT_ATTEMPTED') reEvalStatus = 'INSUFFICIENT_EVIDENCE';

        audit_trail.push({
             decision_id: crypto.randomUUID(),
             claim_id: claim.claim_id,
             candidate_actions: candidates,
             selected_action: selectedAction,
             selection_reason: 'Deterministic planning selected action',
             safety_decision: 'ALLOWED',
             execution_id: `EXE-${crypto.randomUUID()}`,
             result: resultStatus,
             new_evidence: newEvIds,
             re_evaluation_result: reEvalStatus
        });
        
        actionsTaken++;
      }
    }
    
    const { validations, audit_warnings, roadmap } = await ValidationEngine.validate(claims, this.engine.getEvidence(), provider, graph);
    return { validations, audit_warnings, audit_trail, roadmap };
  }
}

import { Claim, ValidationAction } from '../shared/types';
import { GraphNode, EvidenceEdge } from '../evidence/graph-types';
import * as crypto from 'crypto';

export class ValidationPlanner {
  public static plan(
    claim: Claim,
    requirement: string,
    claimSubgraph: { nodes: GraphNode[], edges: EvidenceEdge[], truncated: boolean },
    evaluation: any
  ): ValidationAction[] {
    const actions: ValidationAction[] = [];
    
    if (claim.text.toLowerCase().includes('build')) {
       if (evaluation.execution_status === 'NOT_ATTEMPTED' || evaluation.execution_status === 'ABSENT') {
         actions.push({
          action_id: `ACT-${crypto.randomUUID()}`,
          claim_id: claim.claim_id,
          type: 'BUILD',
          rationale: 'Claim directly requires build verification.',
          required_evidence: ['BUILD_RESULT'],
          risk_level: 'LOW',
          estimated_cost: 'MEDIUM',
          prerequisites: ['SOURCE_CODE']
        });
       }
    }

    if (evaluation.implementation_status === 'PRESENT' || evaluation.implementation_status === 'PARTIAL') {
        if (evaluation.execution_status === 'NOT_ATTEMPTED' || evaluation.execution_status === 'ABSENT') {
            const hasTest = claimSubgraph.nodes.some(n => n.type === 'evidence' && n.metadata && n.metadata.classification === 'TEST');
            
            actions.push({
                action_id: `ACT-${crypto.randomUUID()}`,
                claim_id: claim.claim_id,
                type: 'TEST',
                rationale: 'Claim has implementation evidence but lacks execution/test evidence.',
                required_evidence: ['TEST_RESULT'],
                risk_level: 'LOW',
                estimated_cost: 'MEDIUM',
                prerequisites: ['IMPLEMENTATION']
            });
            
            const testFiles = claimSubgraph.nodes
                .filter(n => n.type === 'evidence' && n.metadata && (n.metadata.location?.includes('.test.') || n.metadata.location?.includes('.spec.')))
                .map(n => n.metadata?.location)
                .filter(Boolean);

                claim_id: claim.claim_id,

                target: testFiles[0],

                required_evidence: ['TEST_RESULT'],
                risk_level: 'LOW',
                estimated_cost: 'LOW',
                prerequisites: ['IMPLEMENTATION']
                });
            }
        }
    }

    // Default to NONE if no action is clearly useful, to prevent unnecessary execution
    if (actions.length === 0) {
       // Only if we need to return something
    }

    return actions;
  }

import { Claim, RemediationTask, DeterministicEvaluation, Evidence } from '../shared/types';
import { GraphNode, EvidenceEdge } from '../evidence/graph-types';
import * as crypto from 'crypto';

export class GraphRoadmapGenerator {
  public static generate(
    claim: Claim,
    requirementId: string | undefined,
    subgraph: { nodes: GraphNode[]; edges: EvidenceEdge[]; truncated: boolean },
    evaluation: DeterministicEvaluation,
    evidenceStore: Evidence[]
  ): RemediationTask[] {
    const tasks: RemediationTask[] = [];
    const evidenceMap = new Map(evidenceStore.map(e => [e.evidence_id, e]));

    // Deterministic roadmap tasks based on constraints and evaluation state
    if (evaluation.implementation_status === 'PARTIAL') {
      tasks.push({
        task_id: `TASK-${crypto.createHash('sha256').update(claim.claim_id + 'PARTIAL_IMPL').digest('hex').substring(0, 8)}`,
        claim_id: claim.claim_id,
        requirement_id: requirementId,
        title: 'Complete partial implementation',
        problem: 'The claim is only partially implemented based on evidence.',
        evidence_refs: subgraph.nodes.map(n => n.id).filter(id => {
            const ev = evidenceMap.get(id);
            return ev && ev.classification === 'IMPLEMENTATION';
        }),
        recommended_actions: ['Implement missing components required by the claim.'],
        acceptance_criteria: ['All required structural evidence (e.g. routes, handlers) is present.']
      });
    }

    if (evaluation.contradiction_status === 'CONTRADICTED' || evaluation.execution_status === 'FAILED' || evaluation.test_status === 'FAILED') {
      tasks.push({
        task_id: `TASK-${crypto.createHash('sha256').update(claim.claim_id + 'EXEC_FAIL').digest('hex').substring(0, 8)}`,
        claim_id: claim.claim_id,
        requirement_id: requirementId,
        title: 'Resolve execution/test failure',
        problem: 'Implementation exists but execution or test failed.',
        evidence_refs: subgraph.nodes.map(n => n.id).filter(id => {
            const ev = evidenceMap.get(id);
            return ev && (ev.classification === 'EXECUTION' || ev.classification === 'TEST');
        }),
        recommended_actions: ['Fix the build or test failure preventing validation.'],
        acceptance_criteria: ['Execution/tests pass successfully.']
      });
    }

    if (evaluation.implementation_status === 'PRESENT' && (evaluation.execution_status === 'ABSENT' || evaluation.execution_status === 'NOT_ATTEMPTED') && (evaluation.test_status === 'ABSENT' || evaluation.test_status === 'NOT_ATTEMPTED')) {
      tasks.push({
        task_id: `TASK-${crypto.createHash('sha256').update(claim.claim_id + 'MISSING_TESTS').digest('hex').substring(0, 8)}`,
        claim_id: claim.claim_id,
        requirement_id: requirementId,
        title: 'Add execution/test coverage',
        problem: 'Implementation exists but has no execution or test evidence.',
        evidence_refs: [],
        recommended_actions: ['Add targeted tests for this specific implementation.'],
        acceptance_criteria: ['A test exists and passes for this feature.']
      });
    }

    // Sort deterministically
    return tasks.sort((a, b) => a.task_id.localeCompare(b.task_id));
  }
}

import { ValidationAction, ValidationActionType } from '../shared/types';
import { SandboxOptions } from '../sandbox/docker';

export interface SafetyDecision {
  status: 'ALLOWED' | 'REJECTED';
  reason?: string;
  sandboxOptions?: SandboxOptions;
  command?: string;
}

export class ValidationSafetyPolicy {
  public static evaluate(action: ValidationAction): SafetyDecision {
    const allowedTypes: ValidationActionType[] = ['STATIC_INSPECTION', 'BUILD', 'TEST', 'TARGETED_TEST', 'HEALTH_CHECK'];
    
    if (!allowedTypes.includes(action.type)) {
      return { status: 'REJECTED', reason: `Action type ${action.type} is not permitted.` };
    }

    // Default strict options based on V0.5 Sandbox restrictions
    const options: SandboxOptions = {
      networkEnabled: false,
      timeoutMs: 30000,
      memoryLimitMb: 512,
      cpus: 1.0,
      pidsLimit: 64
    };

    let command = '';

    switch (action.type) {
      case 'BUILD':
        command = 'npm run build';
        options.timeoutMs = 60000;
        options.memoryLimitMb = 1024;
        break;
      case 'TEST':
        command = 'npm test';
        options.timeoutMs = 60000;
        options.memoryLimitMb = 1024;
        break;
      case 'TARGETED_TEST':
        if (!action.target) return { status: 'REJECTED', reason: 'Targeted test requires a target file or regex' };
        if (/[;&|]/.test(action.target)) {
           return { status: 'REJECTED', reason: 'Target contains shell metacharacters' };
        }
        command = `npm test -- ${action.target}`;
        options.timeoutMs = 60000;
        options.memoryLimitMb = 1024;
        break;
      case 'STATIC_INSPECTION':
        if (!action.target) return { status: 'REJECTED', reason: 'Static inspection requires a target' };
        if (/[;&|]/.test(action.target)) {
           return { status: 'REJECTED', reason: 'Target contains shell metacharacters' };
        }
        command = `cat ${action.target}`;
        break;
      case 'HEALTH_CHECK':
        // For simplicity, we just see if it boots and stops, or run a check script
        command = 'npm run health';
        options.timeoutMs = 15000;
        break;
      default:
        return { status: 'REJECTED', reason: `Unhandled action type: ${action.type}` };
    }

    return {
      status: 'ALLOWED',
      sandboxOptions: options,
      command
    };
  }
}
